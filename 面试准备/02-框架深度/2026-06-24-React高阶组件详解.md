# React 高阶组件（HOC）详解

> 面试重点：HOC 概念、实现方式、应用场景、与 Hooks 的对比

## 一、什么是高阶组件

### 1.1 定义

**高阶组件（Higher-Order Component, HOC）是一个函数，接收一个组件作为参数，返回一个新的增强组件。**

```javascript
// HOC 基本形式
const EnhancedComponent = higherOrderComponent(WrappedComponent)

// 函数签名
type HOC = (WrappedComponent: React.ComponentType) => React.ComponentType
```

**核心特点**：
- 不是组件，是**函数**
- 用于复用组件逻辑
- 不修改原组件，而是**组合**原组件
- 是纯函数，没有副作用

### 1.2 简单示例

```jsx
// 一个简单的 HOC：添加 loading 功能
function withLoading(WrappedComponent) {
  return function EnhancedComponent(props) {
    const [loading, setLoading] = useState(false)
    
    return (
      <div>
        {loading && <div>Loading...</div>}
        <WrappedComponent 
          {...props} 
          loading={loading}
          setLoading={setLoading}
        />
      </div>
    )
  }
}

// 使用
function UserList({ loading, setLoading, users }) {
  useEffect(() => {
    setLoading(true)
    fetchUsers().then(() => setLoading(false))
  }, [])
  
  return <div>{users.map(user => <div key={user.id}>{user.name}</div>)}</div>
}

const UserListWithLoading = withLoading(UserList)
```

### 1.3 命名规范

```javascript
// ✅ 推荐：with 前缀
withRouter
withAuth
withTheme
withLoading

// ✅ 推荐：高阶组件返回的组件使用 displayName
function withLoading(WrappedComponent) {
  const ComponentWithLoading = (props) => {
    // ...
  }
  
  ComponentWithLoading.displayName = 
    `withLoading(${getDisplayName(WrappedComponent)})`
  
  return ComponentWithLoading
}

function getDisplayName(WrappedComponent) {
  return WrappedComponent.displayName || WrappedComponent.name || 'Component'
}
```

## 二、HOC 的两种实现方式

### 2.1 属性代理（Props Proxy）

**最常见的实现方式**：通过 props 传递数据给被包装组件

```javascript
// 1. 操作 props
function withExtraProps(WrappedComponent) {
  return function(props) {
    const extraProps = {
      timestamp: Date.now(),
      version: '1.0.0'
    }
    
    // 合并 props
    return <WrappedComponent {...props} {...extraProps} />
  }
}

// 2. 通过 refs 访问组件实例
function withRef(WrappedComponent) {
  return class extends React.Component {
    componentDidMount() {
      // 可以访问被包装组件的实例
      console.log(this.wrappedInstance)
    }
    
    render() {
      const { forwardedRef, ...rest } = this.props
      
      return (
        <WrappedComponent
          {...rest}
          ref={(instance) => {
            this.wrappedInstance = instance
            if (forwardedRef) {
              forwardedRef(instance)
            }
          }}
        />
      )
    }
  }
}

// 3. 抽象 state
function withToggle(WrappedComponent) {
  return function(props) {
    const [on, setOn] = useState(false)
    
    const toggle = () => setOn(prev => !prev)
    const turnOn = () => setOn(true)
    const turnOff = () => setOn(false)
    
    return (
      <WrappedComponent
        {...props}
        on={on}
        toggle={toggle}
        turnOn={turnOn}
        turnOff={turnOff}
      />
    )
  }
}

// 使用
function Switch({ on, toggle }) {
  return (
    <button onClick={toggle}>
      {on ? 'ON' : 'OFF'}
    </button>
  )
}

const SwitchWithToggle = withToggle(Switch)

// 4. 包装显示内容
function withContainer(WrappedComponent) {
  return function(props) {
    return (
      <div className="container">
        <div className="header">Header</div>
        <WrappedComponent {...props} />
        <div className="footer">Footer</div>
      </div>
    )
  }
}
```

### 2.2 反向继承（Inheritance Inversion）

**通过继承被包装组件实现**：可以访问组件的 state、props、生命周期

```javascript
// 1. 渲染劫持
function withRenderHijack(WrappedComponent) {
  return class extends WrappedComponent {
    render() {
      // 获取原组件的渲染结果
      const tree = super.render()
      
      // 修改渲染结果
      if (tree.type === 'div') {
        const newProps = { ...tree.props, style: { color: 'red' } }
        return React.cloneElement(tree, newProps)
      }
      
      return tree
    }
  }
}

// 2. 操作 state
function withStateLog(WrappedComponent) {
  return class extends WrappedComponent {
    componentDidUpdate(prevProps, prevState) {
      console.log('Previous state:', prevState)
      console.log('Current state:', this.state)
      
      if (super.componentDidUpdate) {
        super.componentDidUpdate(prevProps, prevState)
      }
    }
    
    render() {
      return super.render()
    }
  }
}

// 3. 条件渲染
function withConditionalRender(condition) {
  return function(WrappedComponent) {
    return class extends WrappedComponent {
      render() {
        if (condition(this.props)) {
          return super.render()
        }
        return <div>No permission</div>
      }
    }
  }
}

// 使用
class AdminPanel extends React.Component {
  render() {
    return <div>Admin Panel</div>
  }
}

const AdminPanelWithAuth = withConditionalRender(
  props => props.user?.role === 'admin'
)(AdminPanel)
```

**属性代理 vs 反向继承**：

| 特性 | 属性代理 | 反向继承 |
|------|---------|---------|
| 实现方式 | 函数组件或类组件包装 | 继承被包装组件 |
| 访问 props | ✅ | ✅ |
| 访问 state | ❌ | ✅ |
| 修改渲染结果 | 包装 | 劫持 |
| 生命周期 | 独立 | 共享 |
| 适用范围 | 函数组件 + 类组件 | 仅类组件 |
| 推荐程度 | ⭐⭐⭐⭐⭐ | ⭐⭐ |

## 三、常见 HOC 模式

### 3.1 权限控制

```javascript
// withAuth: 检查用户权限
function withAuth(WrappedComponent, requiredRoles = []) {
  return function AuthComponent(props) {
    const { user } = useAuth()  // 假设有 useAuth hook
    
    if (!user) {
      // 未登录
      return <Redirect to="/login" />
    }
    
    if (requiredRoles.length > 0 && !requiredRoles.includes(user.role)) {
      // 权限不足
      return <div>403 Forbidden</div>
    }
    
    return <WrappedComponent {...props} user={user} />
  }
}

// 使用
const AdminPage = withAuth(AdminPanel, ['admin'])
const UserPage = withAuth(UserProfile, ['user', 'admin'])
```

### 3.2 数据获取

```javascript
// withFetch: 自动获取数据
function withFetch(url, options = {}) {
  return function(WrappedComponent) {
    return function FetchComponent(props) {
      const [data, setData] = useState(null)
      const [loading, setLoading] = useState(true)
      const [error, setError] = useState(null)
      
      useEffect(() => {
        setLoading(true)
        fetch(url, options)
          .then(res => res.json())
          .then(data => {
            setData(data)
            setLoading(false)
          })
          .catch(err => {
            setError(err)
            setLoading(false)
          })
      }, [])
      
      if (loading) return <div>Loading...</div>
      if (error) return <div>Error: {error.message}</div>
      
      return <WrappedComponent {...props} data={data} />
    }
  }
}

// 使用
function UserList({ data }) {
  return (
    <ul>
      {data.map(user => <li key={user.id}>{user.name}</li>)}
    </ul>
  )
}

const UserListWithData = withFetch('/api/users')(UserList)
```

```javascript
// withDebounce: 防抖
function withDebounce(WrappedComponent, delay = 300) {
  return function DebouncedComponent(props) {
    const [debouncedProps, setDebouncedProps] = useState(props)
    
    useEffect(() => {
      const timer = setTimeout(() => {
        setDebouncedProps(props)
      }, delay)
      
      return () => clearTimeout(timer)
    }, [props])
    
    return <WrappedComponent {...debouncedProps} />
  }
}
```

### 3.6 表单增强

```javascript
// withFormField: 表单字段增强
function withFormField(WrappedComponent) {
  return function FormFieldComponent({ name, value, onChange, error, ...rest }) {
    const handleChange = (e) => {
      onChange(name, e.target.value)
    }
    
    return (
      <div className="form-field">
        <WrappedComponent
          {...rest}
          name={name}
          value={value}
          onChange={handleChange}
        />
        {error && <span className="error">{error}</span>}
      </div>
    )
  }
}

// 使用
const Input = ({ name, value, onChange }) => (
  <input name={name} value={value} onChange={onChange} />
)

const FormInput = withFormField(Input)

// 在表单中使用
function LoginForm() {
  const [values, setValues] = useState({ username: '', password: '' })
  const [errors, setErrors] = useState({})
  
  const handleChange = (name, value) => {
    setValues(prev => ({ ...prev, [name]: value }))
  }
  
  return (
    <form>
      <FormInput
        name="username"
        value={values.username}
        onChange={handleChange}
        error={errors.username}
      />
      <FormInput
        name="password"
        type="password"
        value={values.password}
        onChange={handleChange}
        error={errors.password}
      />
    </form>
  )
}
```

### 4.1 多个 HOC 组合

### 4.2 HOC 工厂函数

```javascript
// 带参数的 HOC
function withData(fetchFn, options = {}) {
  return function(WrappedComponent) {
    return function DataComponent(props) {
      const [data, setData] = useState(null)
      const [loading, setLoading] = useState(false)
      const [error, setError] = useState(null)
      
      const loadData = async () => {
        setLoading(true)
        try {
          const result = await fetchFn(props)
          setData(result)
        } catch (err) {
          setError(err)
        } finally {
          setLoading(false)
        }
      }
      
      useEffect(() => {
        if (options.autoLoad !== false) {
          loadData()
        }
      }, [])
      
      return (
        <WrappedComponent
          {...props}
          data={data}
          loading={loading}
          error={error}
          reload={loadData}
        />
      )
    }
  }
}

// 使用
const UserListWithData = withData(
  (props) => fetch(`/api/users?page=${props.page}`).then(r => r.json()),
  { autoLoad: true }
)(UserList)
```

## 五、HOC 的注意事项

### 5.1 不要在 render 中使用 HOC

```javascript
// ❌ 错误：每次渲染都创建新组件
function MyComponent() {
  const EnhancedComponent = withTheme(SomeComponent)
  return <EnhancedComponent />
}

// 问题：
// 1. 每次渲染都会创建新的 EnhancedComponent
// 2. 导致子组件树完全卸载和重新挂载
// 3. 状态丢失，性能问题

// ✅ 正确：在组件外部定义
const EnhancedComponent = withTheme(SomeComponent)

function MyComponent() {
  return <EnhancedComponent />
}
```

### 5.2 传递所有 props

```javascript
// ❌ 错误：没有传递所有 props
function withAuth(WrappedComponent) {
  return function(props) {
    const { user } = useAuth()
    // 只传递了 user，丢失了其他 props
    return <WrappedComponent user={user} />
  }
}

// ✅ 正确：使用展开运算符
function withAuth(WrappedComponent) {
  return function(props) {
    const { user } = useAuth()
    return <WrappedComponent {...props} user={user} />
  }
}
```

### 5.3 复制静态方法

```javascript
// 原组件有静态方法
class MyComponent extends React.Component {
  static myStaticMethod() {
    return 'static method'
  }
  
  render() {
    return <div>MyComponent</div>
  }
}

// ❌ 错误：HOC 不会复制静态方法
function withAuth(WrappedComponent) {
  return class extends React.Component {
    render() {
      return <WrappedComponent {...this.props} />
    }
  }
}

const EnhancedComponent = withAuth(MyComponent)
EnhancedComponent.myStaticMethod  // undefined ❌

// ✅ 方法1：手动复制
function withAuth(WrappedComponent) {
  class Enhanced extends React.Component {
    render() {
      return <WrappedComponent {...this.props} />
    }
  }
  
  // 复制静态方法
  Enhanced.myStaticMethod = WrappedComponent.myStaticMethod
  
  return Enhanced
}

// ✅ 方法2：使用 hoist-non-react-statics
import hoistNonReactStatics from 'hoist-non-react-statics'

function withAuth(WrappedComponent) {
  class Enhanced extends React.Component {
    render() {
      return <WrappedComponent {...this.props} />
    }
  }
  
  // 自动复制所有非 React 静态方法
  hoistNonReactStatics(Enhanced, WrappedComponent)
  
  return Enhanced
}
```

### 5.4 转发 Refs

```javascript
// ❌ 问题：ref 不会被传递
function withAuth(WrappedComponent) {
  return function(props) {
    const { user } = useAuth()
    // ref 是特殊的 prop，不会被传递

function withAuth(WrappedComponent) {
  const Enhanced = React.forwardRef((props, ref) => {
    const { user } = useAuth()
    return <WrappedComponent {...props} user={user} ref={ref} />
  })

  }

function withAuth(WrappedComponent) {
  return function(props) {
    const { innerRef, ...rest } = props
    const { user } = useAuth()
```

## 六、HOC vs Hooks

### 6.1 对比

| 特性 | HOC | Hooks |
|------|-----|-------|
| 逻辑复用 | ✅ | ✅ |
| 嵌套地狱 | ❌ 多个 HOC 会嵌套 | ✅ 扁平化 |
| 命名冲突 | ❌ 容易发生 props 冲突 | ✅ 变量作用域隔离 |
| 可读性 | ❌ 包装层级多 | ✅ 代码更直观 |
| 类型推断 | ❌ TypeScript 复杂 | ✅ 类型友好 |
| 性能 | ❌ 额外组件层级 | ✅ 无额外组件 |
| 学习成本 | 中等 | 较低 |

### 6.2 迁移示例

**HOC 实现**：

```javascript
// 使用 HOC
function withMouse(WrappedComponent) {
  return class extends React.Component {
    state = { x: 0, y: 0 }
    
    handleMouseMove = (e) => {
      this.setState({ x: e.clientX, y: e.clientY })
    }
    
    componentDidMount() {
      window.addEventListener('mousemove', this.handleMouseMove)
    }
    
    componentWillUnmount() {
      window.removeEventListener('mousemove', this.handleMouseMove)
    }
    
    render() {
      return <WrappedComponent {...this.props} mouse={this.state} />
    }
  }
}

const MouseTracker = withMouse(({ mouse }) => (
  <div>
    Mouse position: {mouse.x}, {mouse.y}
  </div>
))
```

**Hooks 实现**：

```javascript
// 使用 Hooks
function useMouse() {
  const [position, setPosition] = useState({ x: 0, y: 0 })
  
  useEffect(() => {
    const handleMouseMove = (e) => {
      setPosition({ x: e.clientX, y: e.clientY })
    }
    
    window.addEventListener('mousemove', handleMouseMove)
    return () => window.removeEventListener('mousemove', handleMouseMove)
  }, [])
  
  return position
}

function MouseTracker() {
  const mouse = useMouse()
  
  return (
    <div>
      Mouse position: {mouse.x}, {mouse.y}
    </div>
  )
}
```

### 6.3 何时使用 HOC？

**仍然适合使用 HOC 的场景**：

1. **第三方库集成**
   ```javascript
   // React Router 的 withRouter
   export default withRouter(MyComponent)
   
   // Redux 的 connect
   export default connect(mapStateToProps)(MyComponent)
   ```

2. **跨多个组件的通用逻辑**
   ```javascript
   // 公司级的权限控制
   export const withCompanyAuth = withAuth({ level: 'company' })
   
   // 多个页面复用
   const AdminPage = withCompanyAuth(AdminPanel)
   const SettingsPage = withCompanyAuth(Settings)
   ```

3. **需要修改组件类型**
   ```javascript
   // 将类组件转为纯组件
   const PureComponent = withPureRender(ClassComponent)
   ```

**推荐使用 Hooks 的场景**：

1. **新项目**：优先使用 Hooks
2. **函数组件**：Hooks 更自然
3. **逻辑组合**：避免嵌套地狱
4. **TypeScript**：类型推断更好

### 7.1 通用错误边界 HOC

```javascript
function withErrorBoundary(WrappedComponent, fallback) {
  return class ErrorBoundary extends React.Component {
    state = { hasError: false, error: null }
    
    static getDerivedStateFromError(error) {
      return { hasError: true, error }
    }
    
    componentDidCatch(error, errorInfo) {
      console.error('Error caught by boundary:', error, errorInfo)
      // 上报错误到监控系统
      reportError(error, errorInfo)
    }
    
    render() {
      if (this.state.hasError) {
        if (fallback) {
          return fallback(this.state.error)
        }
        return <div>Something went wrong.</div>
      }
      
      return <WrappedComponent {...this.props} />
    }
  }
}

// 使用
const SafeUserList = withErrorBoundary(
  UserList,
  (error) => <div>Failed to load users: {error.message}</div>
)
```

### 7.3 响应式 HOC

```javascript
function withResponsive(WrappedComponent) {
  return function ResponsiveComponent(props) {
    const [windowSize, setWindowSize] = useState({
      width: window.innerWidth,
      height: window.innerHeight
    })
    
    useEffect(() => {
      const handleResize = () => {
        setWindowSize({
          width: window.innerWidth,
          height: window.innerHeight
        })
      }
      
      window.addEventListener('resize', handleResize)
      return () => window.removeEventListener('resize', handleResize)
    }, [])
    
    const isMobile = windowSize.width < 768
    const isTablet = windowSize.width >= 768 && windowSize.width < 1024
    const isDesktop = windowSize.width >= 1024
    
    return (
      <WrappedComponent
        {...props}
        windowSize={windowSize}
        isMobile={isMobile}
        isTablet={isTablet}
        isDesktop={isDesktop}
      />
    )
  }
}

// 使用
function Dashboard({ isMobile, isDesktop }) {
  if (isMobile) {
    return <MobileDashboard />
  }
  
  return <DesktopDashboard />
}

const ResponsiveDashboard = withResponsive(Dashboard)
```

## 八、面试高频问题

### Q1: 什么是高阶组件？它解决了什么问题？

**回答要点**：

1. **定义**：接收组件作为参数，返回新组件的函数
2. **目的**：复用组件逻辑，不修改原组件
3. **解决的问题**：
   - 代码复用（避免重复逻辑）
   - 关注点分离（业务逻辑与展示分离）
   - 增强组件功能（权限、日志、性能监控等）

```javascript
// 示例
const EnhancedComponent = withAuth(MyComponent)

// 等价于
function EnhancedComponent(props) {
  // 添加认证逻辑
  const user = useAuth()
  if (!user) return <Login />
  
  // 渲染原组件
  return <MyComponent {...props} user={user} />
}
```

### Q2: HOC 和 Hooks 有什么区别？

**核心区别**：

1. **形式**
   - HOC：组件包装组件
   - Hooks：函数调用函数

2. **组件层级**
   - HOC：增加组件层级
   - Hooks：不增加层级

3. **props 冲突**
   - HOC：容易发生 props 命名冲突
   - Hooks：变量作用域隔离

4. **可读性**
   - HOC：多层嵌套难以追踪
   - Hooks：扁平化，更直观

```javascript
// HOC: 3层嵌套
<WithAuth>
  <WithTheme>
    <WithData>
      <MyComponent />
    </WithData>
  </WithTheme>
</WithAuth>

// Hooks: 扁平化
function MyComponent() {
  const auth = useAuth()
  const theme = useTheme()
  const data = useData()
  // ...
}
```

### Q3: HOC 的缺点是什么？

1. **嵌套地狱**：多个 HOC 导致深层嵌套
2. **Props 冲突**：多个 HOC 可能注入同名 props
3. **调试困难**：React DevTools 显示多层包装组件
4. **静态方法丢失**：需要手动复制
5. **Ref 不透传**：需要使用 forwardRef
6. **性能开销**：额外的组件层级

### Q4: 如何避免 HOC 的命名冲突？

**解决方案**：

```javascript
// 1. 使用命名空间
function withTheme(WrappedComponent) {
  return function(props) {
    const theme = useTheme()
    
    // 使用命名空间
    return <WrappedComponent {...props} theme_data={theme} />
  }
}

// 2. 使用对象包装
function withMouse(WrappedComponent) {
  return function(props) {
    const position = useMouse()
    
    // 包装成对象
    return <WrappedComponent {...props} mouse={{ position }} />
  }
}

// 3. 使用 Render Props
function withData(WrappedComponent) {
  return function(props) {
    const data = useData()
    
    // 通过 render prop 传递
    return <WrappedComponent {...props}>{data}</WrappedComponent>
  }
}
```

### Q5: 能否在 render 方法中使用 HOC？

**❌ 不能！**

```javascript
// ❌ 错误
function MyComponent() {
  const EnhancedComponent = withAuth(SomeComponent)
  return <EnhancedComponent />
}

// 问题：
// 1. 每次渲染都创建新组件
// 2. 新组件 !== 旧组件
// 3. React 认为是不同组件，会完全卸载旧的，挂载新的
// 4. 导致状态丢失、性能问题

// ✅ 正确
const EnhancedComponent = withAuth(SomeComponent)

function MyComponent() {
  return <EnhancedComponent />
}
```

### Q6: HOC 和装饰器（Decorator）的关系？

**HOC 可以用装饰器语法**：

```javascript
// 装饰器语法（需要 Babel 支持）
@withAuth
@withTheme
@withLogger
class MyComponent extends React.Component {
  render() {
    return <div>MyComponent</div>
  }
}

// 等价于
const MyComponent = withAuth(
  withTheme(
    withLogger(
      class extends React.Component {
        render() {
          return <div>MyComponent</div>
        }
      }
    )
  )
)

// 注意：
// 1. 装饰器只是语法糖，本质还是 HOC
// 2. 装饰器从下往上执行
// 3. 目前仍是提案阶段（Stage 3）
```

## 九、总结

### 核心要点

1. **定义**：HOC 是参数为组件，返回值为新组件的函数
2. **实现方式**：属性代理（推荐）、反向继承
3. **应用场景**：权限控制、数据获取、性能优化、日志记录等
4. **注意事项**：不在 render 中创建、传递所有 props、复制静态方法、转发 refs
5. **现代替代**：Hooks 是更好的选择，但 HOC 在某些场景仍有价值

### 使用建议

1. **新项目**：优先使用 Hooks
2. **旧项目**：可以继续使用 HOC，逐步迁移
3. **第三方库集成**：HOC 仍是主流方式
4. **复杂逻辑**：考虑 Render Props 或 Hooks

### 学习路径

1. **基础**：理解 HOC 概念和基本用法
2. **进阶**：掌握两种实现方式的区别
3. **实战**：实现常见的 HOC（权限、日志、性能监控）
4. **对比**：了解 HOC、Render Props、Hooks 的优劣
