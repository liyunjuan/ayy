## Electron简介

### 什么是Electron？

Electron是一个使用JavaScript、HTML和CSS构建跨平台桌面应用的框架。

**核心特点：**
- ✅ 基于Chromium和Node.js
- ✅ 跨平台（Windows、macOS、Linux）
- ✅ 使用Web技术开发
- ✅ 可访问原生API
- ✅ 自动更新
- ✅ 丰富的生态

**知名应用：**
- VS Code（代码编辑器）
- Slack（团队协作）
- Discord（即时通讯）
- Figma Desktop（设计工具）
- Notion（笔记应用）
- Postman（API测试）

### 技术架构

```
Electron应用
    ├─ Chromium（渲染引擎）
    │   └─ 渲染进程（Renderer Process）
    │       └─ Web页面（HTML/CSS/JS）
    │
    ├─ Node.js（后端运行时）
    │   └─ 主进程（Main Process）
    │       └─ 应用逻辑、原生API
    │
    └─ Native APIs
        ├─ 文件系统
        ├─ 系统托盘
        ├─ 通知
        └─ 原生对话框
```

## 核心概念

### 1. 主进程（Main Process）

**特点：**
- 每个Electron应用只有一个主进程
- 运行在Node.js环境
- 可以访问所有Node.js API
- 管理应用生命周期
- 创建和管理窗口

**职责：**
```javascript
// 主进程职责
- 创建窗口（BrowserWindow）
- 管理应用生命周期（app）
- 访问原生API（Menu、Tray、Dialog等）
- 管理全局状态
- 与渲染进程通信
```

### 2. 渲染进程（Renderer Process）

**特点：**
- 每个BrowserWindow对应一个渲染进程
- 运行在Chromium环境
- 类似Web页面
- 可以使用Web API
- 通过IPC与主进程通信

**职责：**
```javascript
// 渲染进程职责
- 显示UI界面
- 处理用户交互
- 执行Web代码（HTML/CSS/JS）
- 与主进程通信
```

### 3. 预加载脚本（Preload Script）

**作用：**
- 在渲染进程加载前执行
- 桥接主进程和渲染进程
- 安全地暴露API给渲染进程

**示例：**
```javascript
// preload.js
const { contextBridge, ipcRenderer } = require('electron');

// 暴露API给渲染进程
contextBridge.exposeInMainWorld('electronAPI', {
  // 发送消息到主进程
  sendMessage: (message) => ipcRenderer.send('message', message),
  
  // 接收主进程消息
  onReply: (callback) => ipcRenderer.on('reply', callback),
  
  // 双向通信
  invoke: (channel, data) => ipcRenderer.invoke(channel, data),
});

// 渲染进程中使用
// window.electronAPI.sendMessage('Hello');
```

### 主进程代码

```javascript
// main.js
const { app, BrowserWindow } = require('electron');
const path = require('path');

let mainWindow;

// 创建主窗口
function createWindow() {
  mainWindow = new BrowserWindow({
    width: 1200,
    height: 800,
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      nodeIntegration: false, // 安全：禁用Node集成
      contextIsolation: true,  // 安全：启用上下文隔离
    },
  });

  // 加载页面
  mainWindow.loadFile('index.html');
  
  // 开发环境打开DevTools
  if (process.env.NODE_ENV === 'development') {
    mainWindow.webContents.openDevTools();
  }
  
  // 窗口关闭事件
  mainWindow.on('closed', () => {
    mainWindow = null;
  });
}

// 应用就绪时创建窗口
app.whenReady().then(() => {
  createWindow();
  
  // macOS：点击Dock图标时重新创建窗口
  app.on('activate', () => {
    if (BrowserWindow.getAllWindows().length === 0) {
      createWindow();
    }
  });
});

// 所有窗口关闭时退出应用（macOS除外）
app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') {
    app.quit();
  }
});
```

### 预加载脚本

```javascript
// preload.js
const { contextBridge, ipcRenderer } = require('electron');

contextBridge.exposeInMainWorld('electronAPI', {
  platform: process.platform,
  versions: process.versions,
  
  // IPC通信
  send: (channel, data) => {
    const validChannels = ['toMain'];
    if (validChannels.includes(channel)) {
      ipcRenderer.send(channel, data);
    }
  },
  
  receive: (channel, func) => {
    const validChannels = ['fromMain'];
    if (validChannels.includes(channel)) {
      ipcRenderer.on(channel, (event, ...args) => func(...args));
    }
  },
  
  invoke: (channel, data) => {
    const validChannels = ['getData', 'saveData'];
    if (validChannels.includes(channel)) {
      return ipcRenderer.invoke(channel, data);
    }
  },
});
```

## 进程间通信IPC

### 1. 单向通信（Renderer → Main）

```javascript
// preload.js
const { contextBridge, ipcRenderer } = require('electron');

contextBridge.exposeInMainWorld('electronAPI', {
  sendMessage: (message) => ipcRenderer.send('message', message),
});

// renderer.js
window.electronAPI.sendMessage('Hello from renderer');

// main.js
const { ipcMain } = require('electron');

ipcMain.on('message', (event, message) => {
  console.log('收到消息:', message);
});
```

### 2. 双向通信（Renderer ↔ Main）

**方式1：send + on**

```javascript
// preload.js
contextBridge.exposeInMainWorld('electronAPI', {
  sendRequest: (data) => ipcRenderer.send('request', data),
  onResponse: (callback) => ipcRenderer.on('response', (event, data) => callback(data)),
});

// renderer.js
window.electronAPI.sendRequest({ query: 'data' });
window.electronAPI.onResponse((data) => {
  console.log('收到响应:', data);
});

// main.js
ipcMain.on('request', (event, data) => {
  console.log('收到请求:', data);
  
  // 回复发送者
  event.sender.send('response', { result: 'success' });
});
```

**方式2：invoke + handle（推荐）**

```javascript
// preload.js
contextBridge.exposeInMainWorld('electronAPI', {
  getData: () => ipcRenderer.invoke('get-data'),
  saveData: (data) => ipcRenderer.invoke('save-data', data),
});

// renderer.js
async function fetchData() {
  try {
    const data = await window.electronAPI.getData();
    console.log('获取数据:', data);
  } catch (error) {
    console.error('获取失败:', error);
  }
}

async function saveData() {
  try {
    const result = await window.electronAPI.saveData({ name: 'John' });
    console.log('保存结果:', result);
  } catch (error) {
    console.error('保存失败:', error);
  }
}

// main.js
ipcMain.handle('get-data', async (event) => {
  // 异步获取数据
  const data = await fetchDataFromDB();
  return data;
});

ipcMain.handle('save-data', async (event, data) => {
  // 异步保存数据
  await saveDataToDB(data);
  return { success: true };
});
```

### 3. 主进程主动发送消息

```javascript
// main.js
const { BrowserWindow } = require('electron');

function sendToRenderer(channel, data) {
  const win = BrowserWindow.getFocusedWindow();
  if (win) {
    win.webContents.send(channel, data);
  }
}

// 发送消息
sendToRenderer('notification', { message: 'Update available!' });

// preload.js
contextBridge.exposeInMainWorld('electronAPI', {
  onNotification: (callback) => {
    ipcRenderer.on('notification', (event, data) => callback(data));
  },
});

// renderer.js
window.electronAPI.onNotification((data) => {
  console.log('通知:', data);
});
```

### 4. 渲染进程之间通信

```javascript
// 方式1：通过主进程中转
// renderer1.js
window.electronAPI.sendToOther({ from: 'window1', message: 'Hello' });

// main.js
ipcMain.on('send-to-other', (event, data) => {
  // 转发给其他窗口
  BrowserWindow.getAllWindows().forEach((win) => {
    if (win.webContents !== event.sender) {
      win.webContents.send('message-from-other', data);
    }
  });
});

// renderer2.js
window.electronAPI.onMessageFromOther((data) => {
  console.log('收到其他窗口消息:', data);
});

// 方式2：使用MessagePort（推荐）
// main.js
const { MessageChannelMain } = require('electron');

const { port1, port2 } = new MessageChannelMain();

// 将port1发送给窗口1
win1.webContents.postMessage('port', null, [port1]);

// 将port2发送给窗口2
win2.webContents.postMessage('port', null, [port2]);

// 窗口1和窗口2可以直接通信
```

```javascript
    { type: 'separator' },
```

```javascript
        app.quit();
      },
    },

  tray.on('click', () => {
    mainWindow.isVisible() ? mainWindow.hide() : mainWindow.show();

  // 双击托盘图标
  tray.on('double-click', () => {
    mainWindow.show();
  });
});
```

### 案例1：简单的文本编辑器

```javascript
// main.js
const { app, BrowserWindow, dialog, ipcMain } = require('electron');
const fs = require('fs');

let mainWindow;
let currentFile = null;

function createWindow() {
  mainWindow = new BrowserWindow({
    width: 1000,
    height: 700,
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      nodeIntegration: false,
      contextIsolation: true,
    },
  });
  
  mainWindow.loadFile('index.html');
}

// 打开文件
ipcMain.handle('open-file', async () => {
  const result = await dialog.showOpenDialog({
    properties: ['openFile'],
    filters: [
      { name: 'Text Files', extensions: ['txt', 'md'] },
      { name: 'All Files', extensions: ['*'] },
    ],
  });
  
  if (!result.canceled && result.filePaths.length > 0) {
    const filePath = result.filePaths[0];
    const content = fs.readFileSync(filePath, 'utf-8');
    currentFile = filePath;
    return { filePath, content };
  }
  
  return null;
});

// 保存文件
ipcMain.handle('save-file', async (event, content) => {
  if (currentFile) {
    fs.writeFileSync(currentFile, content, 'utf-8');
    return { success: true, filePath: currentFile };
  } else {
    const result = await dialog.showSaveDialog({
      filters: [
        { name: 'Text Files', extensions: ['txt'] },
      ],
    });
    
    if (!result.canceled) {
      fs.writeFileSync(result.filePath, content, 'utf-8');
      currentFile = result.filePath;
      return { success: true, filePath: result.filePath };
    }
  }
  
  return { success: false };
});

app.whenReady().then(createWindow);
```

### 案例2：系统托盘应用

```javascript
// main.js - 最小化到托盘
const { app, BrowserWindow, Tray, Menu, nativeImage } = require('electron');

let mainWindow;
let tray;

function createWindow() {
  mainWindow = new BrowserWindow({
    width: 800,
    height: 600,
    show: false,
  });
  
  mainWindow.loadFile('index.html');
  
  mainWindow.on('ready-to-show', () => {
    mainWindow.show();
  });
  
  // 关闭时最小化到托盘
  mainWindow.on('close', (event) => {
    if (!app.isQuitting) {
      event.preventDefault();
      mainWindow.hide();
    }
  });
}

function createTray() {
  const icon = nativeImage.createFromPath(path.join(__dirname, 'tray.png'));
  tray = new Tray(icon);
  
  const contextMenu = Menu.buildFromTemplate([
    {
      label: '显示',
      click: () => mainWindow.show(),
    },
    {
      label: '退出',
      click: () => {
        app.isQuitting = true;
        app.quit();
      },
    },
  ]);
  
  tray.setContextMenu(contextMenu);
  tray.setToolTip('My App');
  
  tray.on('click', () => {
    mainWindow.isVisible() ? mainWindow.hide() : mainWindow.show();
  });
}

app.whenReady().then(() => {
  createWindow();
  createTray();
});

app.on('before-quit', () => {
  app.isQuitting = true;
});
```

## 面试高频问题

### Q1: Electron的架构是怎样的？

**答案：**

Electron采用**多进程架构**，包含：

**1. 主进程（Main Process）**
- 每个应用只有1个
- 运行在Node.js环境
- 管理应用生命周期
- 创建和管理窗口
- 访问原生API

**2. 渲染进程（Renderer Process）**
- 每个BrowserWindow一个
- 运行在Chromium环境
- 显示Web页面
- 处理UI交互

**3. 预加载脚本（Preload Script）**
- 桥接主进程和渲染进程
- 安全地暴露API

**通信方式：**
```javascript
// IPC通信
// 渲染进程 → 主进程：ipcRenderer.send / invoke
// 主进程 → 渲染进程：webContents.send
// 双向通信：invoke + handle（推荐）
```

**安全模型：**
```javascript
webPreferences: {
  nodeIntegration: false,      // 禁用Node集成
  contextIsolation: true,       // 启用上下文隔离
  preload: path.join(__dirname, 'preload.js'),
}
```
