## 二、Fiber 架构

### 2.1 为什么需要 Fiber？

**React 15 的问题（Stack Reconciler）**：

```javascript
// React 15: 递归更新，不可中断
function mountComponent(vnode) {
  const instance = new vnode.type(vnode.props)
  const rendered = instance.render()
  
  // 递归挂载子组件
  rendered.children.forEach(child => {
    mountComponent(child) // 无法中断
  })
  
  // 问题：
  // 1. JS 执行长时间占用主线程
  // 2. 导致页面卡顿、掉帧
  // 3. 用户交互响应慢
}
```

**Fiber 的解决方案**：

1. **可中断**：将递归改为循环，可以随时暂停和恢复
2. **优先级**：高优先级任务优先执行
3. **时间切片**：将工作分片，每片 5ms，避免长时间阻塞

### 2.2 Fiber 节点结构

```javascript
// packages/react-reconciler/src/ReactFiber.js
function FiberNode(tag, pendingProps, key, mode) {
  // ===== 实例属性 =====
  this.tag = tag                  // Fiber 类型（函数组件、类组件等）
  this.key = key                  // key 属性
  this.elementType = null         // React 元素类型
  this.type = null                // 函数或类
  this.stateNode = null           // 对应的 DOM 节点或组件实例

  // ===== Fiber 树结构 =====
  this.return = null              // 父 Fiber
  this.child = null               // 第一个子 Fiber
  this.sibling = null             // 下一个兄弟 Fiber
  this.index = 0                  // 在父 Fiber 中的索引

  // ===== 状态相关 =====
  this.pendingProps = pendingProps  // 新的 props
  this.memoizedProps = null         // 上次渲染的 props
  this.updateQueue = null           // 更新队列
  this.memoizedState = null         // 上次渲染的 state
  this.dependencies = null          // 依赖的 context、订阅等

  // ===== 副作用 =====
  this.flags = NoFlags              // 副作用标记（增、删、改）
  this.subtreeFlags = NoFlags       // 子树副作用标记
  this.deletions = null             // 要删除的子 Fiber

  // ===== 调度相关 =====
  this.lanes = NoLanes              // 优先级
  this.childLanes = NoLanes         // 子树优先级

  // ===== 双缓存 =====
  this.alternate = null             // 指向另一棵树的对应节点
}

// Fiber 类型
const FunctionComponent = 0       // 函数组件
const ClassComponent = 1          // 类组件
const HostRoot = 3                // 根节点
const HostComponent = 5           // 原生 DOM 组件
const HostText = 6                // 文本节点

// 副作用标记
const Placement = 0b00010         // 插入
const Update = 0b00100            // 更新
const Deletion = 0b01000          // 删除
const ChildDeletion = 0b10000     // 子节点删除
```

### 2.3 双缓存机制

```javascript
// React 同时维护两棵 Fiber 树
// - current tree: 当前屏幕显示的内容
// - workInProgress tree: 正在构建的新树

// 首次渲染
const root = createRoot(document.getElementById('root'))
root.render(<App />)

// 过程：
// 1. 创建 FiberRoot 和 rootFiber
const fiberRoot = {
  current: rootFiber,     // 指向 current 树的根
  // ...
}

const rootFiber = {
  stateNode: fiberRoot,   // 指向 FiberRoot
  alternate: null,        // 首次渲染为 null
  // ...
}

// 2. 基于 current 树创建 workInProgress 树
function createWorkInProgress(current, pendingProps) {
  let workInProgress = current.alternate
  
  if (workInProgress === null) {
    // 首次渲染：创建新 Fiber
    workInProgress = createFiber(
      current.tag,
      pendingProps,
      current.key,
      current.mode
    )
    workInProgress.alternate = current
    current.alternate = workInProgress
  } else {
    // 更新：复用 alternate
    workInProgress.pendingProps = pendingProps
    workInProgress.flags = NoFlags
    workInProgress.subtreeFlags = NoFlags
    workInProgress.deletions = null
  }
  
  // 复用属性
  workInProgress.child = current.child
  workInProgress.memoizedProps = current.memoizedProps
  workInProgress.memoizedState = current.memoizedState
  // ...
  
  return workInProgress
}

// 3. commit 阶段：交换 current 指针
fiberRoot.current = workInProgress  // workInProgress 变成 current

// 下次更新时，原来的 current 变成新的 workInProgress
```

**双缓存的优势**：

```javascript
// 类似帧缓冲技术
// 前台显示 current 树，后台构建 workInProgress 树
// 构建完成后一次性切换，避免白屏或闪烁

function doubleBuffer() {
  let showing = 'buffer1'

  // 用户看到的是 buffer1
  display(buffer1)
  
  // 后台在 buffer2 上渲染

  // 渲染完成，瞬间切换
```

## 三、调度机制（Scheduler）

### 3.1 优先级体系

```javascript
// packages/scheduler/src/SchedulerPriorities.js

// Scheduler 优先级（5种）
export const ImmediatePriority = 1      // 立即执行（最高优先级）
export const UserBlockingPriority = 2   // 用户阻塞（用户交互）
export const NormalPriority = 3         // 正常优先级
export const LowPriority = 4            // 低优先级
export const IdlePriority = 5           // 空闲优先级（最低）

// React Lane 优先级（31条赛道）
export const SyncLane = 0b0000000000000000000000000001      // 同步
export const InputContinuousLane = 0b0000000000000000000000000100  // 连续输入
export const DefaultLane = 0b0000000000000000000000010000    // 默认
export const TransitionLane1 = 0b0000000000000000001000000000 // 过渡

// 优先级转换
function schedulerPriorityToLanePriority(schedulerPriority) {
  switch (schedulerPriority) {
    case ImmediatePriority:
      return SyncLane
    case UserBlockingPriority:
      return InputContinuousLane
    case NormalPriority:
    default:
      return DefaultLane
    case LowPriority:
      return TransitionLane1
    case IdlePriority:
      return IdleLane
  }
}
```

**不同场景的优先级**：

```javascript
// 1. 同步优先级：立即执行
ReactDOM.flushSync(() => {
  setState(newState)  // SyncLane
})

// 2. 用户交互：高优先级
<input onChange={e => setState(e.target.value)} />  // InputContinuousLane

// 3. 普通更新：正常优先级
useEffect(() => {
  setState(newState)  // DefaultLane
}, [])

// 4. startTransition：低优先级
startTransition(() => {
  setState(newState)  // TransitionLane
})

// 5. useDeferredValue：延迟更新
const deferredValue = useDeferredValue(value)  // TransitionLane
```

### 3.2 时间切片

```javascript
// packages/scheduler/src/forks/Scheduler.js

// 每帧时间预算：5ms
let frameInterval = 5

// 任务调度核心
function workLoop(hasTimeRemaining, initialTime) {
  let currentTime = initialTime
  
  // 从任务队列取出最高优先级任务
  currentTask = peek(taskQueue)
  
  while (currentTask !== null) {
    if (
      currentTask.expirationTime > currentTime &&
      (!hasTimeRemaining || shouldYieldToHost())  // 判断是否需要让出主线程
    ) {
      // 时间片用完，退出循环
      break
    }
    
    // 执行任务
    const callback = currentTask.callback
    if (typeof callback === 'function') {
      const continuationCallback = callback(
        currentTask.expirationTime <= currentTime  // 是否过期
      )
      
      if (typeof continuationCallback === 'function') {
        // 任务未完成，保存继续执行的函数
        currentTask.callback = continuationCallback
      } else {
        // 任务完成，从队列移除
        if (currentTask === peek(taskQueue)) {
          pop(taskQueue)
        }
      }
    }
    
    currentTask = peek(taskQueue)
  }
  
  // 还有任务，请求下一帧继续执行
  if (currentTask !== null) {
    return true
  }
  
  return false
}

// 判断是否需要让出主线程
function shouldYieldToHost() {
  const timeElapsed = getCurrentTime() - startTime
  if (timeElapsed < frameInterval) {
    // 还有时间，继续执行
    return false
  }
  // 时间片用完，让出主线程
  return true
}

// 请求调度
function requestHostCallback(callback) {
  scheduledHostCallback = callback
  
  if (!isMessageLoopRunning) {
    isMessageLoopRunning = true
    // 使用 MessageChannel 模拟宏任务
    schedulePerformWorkUntilDeadline()
  }
}

// MessageChannel 实现
const channel = new MessageChannel()
const port = channel.port2
channel.port1.onmessage = performWorkUntilDeadline

function schedulePerformWorkUntilDeadline() {
  port.postMessage(null)
}
```

**为什么用 MessageChannel 而不是 setTimeout？**

```javascript
// setTimeout 最小延迟 4ms，且嵌套层级 >5 时会强制 4ms
// MessageChannel 更快，且不受嵌套层级限制

// 测试对比
console.time('setTimeout')
setTimeout(() => console.timeEnd('setTimeout'), 0)  // ~4ms

console.time('MessageChannel')
const channel = new MessageChannel()
channel.port1.onmessage = () => console.timeEnd('MessageChannel')
channel.port2.postMessage(null)  // ~0.1ms
```

## 四、更新流程

### 4.1 Render 阶段（可中断）

```javascript
// packages/react-reconciler/src/ReactFiberWorkLoop.js

function performConcurrentWorkOnRoot(root) {
  // 1. 获取本次更新的优先级
  const lanes = getNextLanes(root, NoLanes)
  
  // 2. 开始 render 阶段
  let exitStatus = renderRootConcurrent(root, lanes)
  
  // 3. 进入 commit 阶段
  if (exitStatus === RootCompleted) {
    commitRoot(root)
  }
}

function renderRootConcurrent(root, lanes) {
  // 准备工作：创建 workInProgress 树
  prepareFreshStack(root, lanes)
  
  // 循环构建 Fiber 树
  do {
    try {
      workLoopConcurrent()
      break
    } catch (thrownValue) {
      handleError(root, thrownValue)
    }
  } while (true)
  
  return workInProgressRootExitStatus
}

function workLoopConcurrent() {
  // 可中断的循环
  while (workInProgress !== null && !shouldYield()) {
    performUnitOfWork(workInProgress)
  }
}

function performUnitOfWork(unitOfWork) {
  const current = unitOfWork.alternate
  
  // 1. beginWork: 向下遍历
  let next = beginWork(current, unitOfWork, renderLanes)
  
  unitOfWork.memoizedProps = unitOfWork.pendingProps
  
  if (next === null) {
    // 没有子节点，开始 completeWork
    completeUnitOfWork(unitOfWork)
  } else {
    // 继续处理子节点
    workInProgress = next
  }
}
```

**beginWork 流程**：

```javascript
// packages/react-reconciler/src/ReactFiberBeginWork.js

function beginWork(current, workInProgress, renderLanes) {
  // 根据 Fiber 类型，进入不同的处理逻辑
  switch (workInProgress.tag) {
    case FunctionComponent: {
      const Component = workInProgress.type
      const unresolvedProps = workInProgress.pendingProps
      return updateFunctionComponent(
        current,
        workInProgress,
        Component,
        unresolvedProps,
        renderLanes
      )
    }
    case ClassComponent: {
      const Component = workInProgress.type
      const unresolvedProps = workInProgress.pendingProps
      return updateClassComponent(
        current,
        workInProgress,
        Component,
        unresolvedProps,
        renderLanes
      )
    }
    case HostComponent: {
      return updateHostComponent(current, workInProgress, renderLanes)
    }
    case HostText: {
      return null  // 文本节点没有子节点
    }
  }
}

// 函数组件更新
function updateFunctionComponent(
  current,
  workInProgress,
  Component,
  nextProps,
  renderLanes
) {
  // 1. 执行函数组件，获取 children
  let nextChildren = renderWithHooks(
    current,
    workInProgress,
    Component,
    nextProps,
    context,
    renderLanes
  )
  
  // 2. 协调子节点（Diff 算法）
  reconcileChildren(current, workInProgress, nextChildren, renderLanes)
  
  // 3. 返回第一个子 Fiber
  return workInProgress.child
}

// 协调子节点
function reconcileChildren(current, workInProgress, nextChildren, lanes) {
  if (current === null) {
    // 首次渲染：mount
    workInProgress.child = mountChildFibers(
      workInProgress,
      null,
      nextChildren,
      lanes
    )
  } else {
    // 更新：Diff
    workInProgress.child = reconcileChildFibers(
      workInProgress,
      current.child,
      nextChildren,
      lanes
    )
  }
}
```

**completeWork 流程**：

```javascript
// packages/react-reconciler/src/ReactFiberCompleteWork.js

function completeUnitOfWork(unitOfWork) {
  let completedWork = unitOfWork
  
  do {
    const current = completedWork.alternate
    const returnFiber = completedWork.return
    
    // 1. completeWork: 处理当前节点
    completeWork(current, completedWork, renderLanes)
    
    // 2. 收集副作用（React 18 使用 subtreeFlags）
    const siblingFiber = completedWork.sibling
    if (siblingFiber !== null) {
      // 有兄弟节点，处理兄弟节点
      workInProgress = siblingFiber
      return
    }
    
    // 3. 返回父节点
    completedWork = returnFiber
    workInProgress = completedWork
  } while (completedWork !== null)
}

function completeWork(current, workInProgress, renderLanes) {
  const newProps = workInProgress.pendingProps
  
  switch (workInProgress.tag) {
    case HostComponent: {
      if (current !== null && workInProgress.stateNode != null) {
        // 更新：标记 Update flag
        updateHostComponent(
          current,
          workInProgress,
          workInProgress.type,
          newProps
        )
      } else {
        // 首次渲染：创建 DOM
        const type = workInProgress.type
        const instance = createInstance(type, newProps)
        
        // 将子 DOM 插入
        appendAllChildren(instance, workInProgress)
        
        // 保存 DOM 引用
        workInProgress.stateNode = instance
        
        // 标记 Placement flag
        workInProgress.flags |= Placement
      }
      return null
    }
  }
}

// 创建 DOM
function createInstance(type, props) {
  const domElement = document.createElement(type)
  // 设置属性
  for (const propKey in props) {
    if (propKey === 'children') continue
    if (propKey === 'style') {
      setStyle(domElement, props.style)
    } else {
      domElement.setAttribute(propKey, props[propKey])
    }
  }
  return domElement
}
```

### 4.2 Commit 阶段（不可中断）

```javascript
// packages/react-reconciler/src/ReactFiberCommitWork.js

function commitRoot(root) {
  const finishedWork = root.finishedWork
  
  // Commit 阶段分为三个子阶段
  commitRootImpl(root, finishedWork)
}

function commitRootImpl(root, finishedWork) {
  // ===== 阶段1: Before Mutation =====
  // 执行 getSnapshotBeforeUpdate
  commitBeforeMutationEffects(root, finishedWork)
  
  // ===== 阶段2: Mutation =====
  // 操作 DOM（增删改）
  commitMutationEffects(root, finishedWork)
  
  // 切换 current 指针
  root.current = finishedWork
  
  // ===== 阶段3: Layout =====
  // 执行 componentDidMount/componentDidUpdate
  // 执行 useLayoutEffect
  commitLayoutEffects(finishedWork, root)
  
  // ===== 异步调度 useEffect =====
  scheduleCallback(NormalPriority, () => {
    flushPassiveEffects()  // 执行 useEffect
  })
}
```

**Mutation 阶段**：

```javascript
function commitMutationEffects(root, finishedWork) {
  // 深度优先遍历
  commitMutationEffectsOnFiber(finishedWork, root)
}

function commitMutationEffectsOnFiber(finishedWork, root) {
  const flags = finishedWork.flags
  
  // 1. 处理 ContentReset（重置文本节点）
  if (flags & ContentReset) {
    commitResetTextContent(finishedWork)
  }
  
  // 2. 处理 Ref
  if (flags & Ref) {
    const current = finishedWork.alternate
    if (current !== null) {
      commitDetachRef(current)
    }
  }
  
  // 3. 处理 Placement（插入）
  if (flags & Placement) {
    commitPlacement(finishedWork)
    finishedWork.flags &= ~Placement  // 清除标记
  }
  
  // 4. 处理 Update（更新）
  if (flags & Update) {
    const current = finishedWork.alternate
    commitWork(current, finishedWork)
  }
  
  // 5. 处理 Deletion（删除）
  if (flags & ChildDeletion) {
    const deletions = finishedWork.deletions
    if (deletions !== null) {
      for (let i = 0; i < deletions.length; i++) {
        const childToDelete = deletions[i]
        commitDeletion(root, childToDelete)
      }
    }
  }
}

// 插入 DOM
function commitPlacement(finishedWork) {
  // 1. 找到父 DOM 节点
  const parentFiber = getHostParentFiber(finishedWork)
  const parentDOM = parentFiber.stateNode
  
  // 2. 找到插入位置（before）
  const before = getHostSibling(finishedWork)
  
  // 3. 插入 DOM
  insertOrAppendPlacementNode(finishedWork, before, parentDOM)
}

function insertOrAppendPlacementNode(node, before, parent) {
  const tag = node.tag
  const isHost = tag === HostComponent || tag === HostText
  
  if (isHost) {
    const stateNode = node.stateNode
    if (before) {
      parent.insertBefore(stateNode, before)
    } else {
      parent.appendChild(stateNode)
    }
  } else {
    // 不是 Host 组件，递归处理子节点
    const child = node.child
    if (child !== null) {
      insertOrAppendPlacementNode(child, before, parent)
      let sibling = child.sibling
      while (sibling !== null) {
        insertOrAppendPlacementNode(sibling, before, parent)
        sibling = sibling.sibling
      }
    }
  }
}
```

## 五、Diff 算法

### 5.1 Diff 原则

```javascript
// React Diff 三大策略：
// 1. 只对同层级节点进行比较（Tree Diff）
// 2. 不同类型的元素会产生不同的树（Component Diff）
// 3. 通过 key 标识哪些子元素是稳定的（Element Diff）

// 示例：
// 旧树                    新树
//   A                      A
//  / \                    / \
// B   C        =>        D   C
//    / \                    / \
//   D   E                  B   E

// React 不会检测到 B 移动到 D 下面
// 而是删除 B，创建新的 D，复用 C
```

### 5.2 单节点 Diff

```javascript
// packages/react-reconciler/src/ReactChildFiber.js

function reconcileSingleElement(
  returnFiber,
  currentFirstChild,
  element,
  lanes
) {
  const key = element.key
  let child = currentFirstChild
  
  // 遍历旧的子节点
  while (child !== null) {
    // 1. 比较 key
    if (child.key === key) {
      // 2. 比较 type
      if (child.elementType === element.type) {
        // key 和 type 都相同，复用节点
        deleteRemainingChildren(returnFiber, child.sibling)
        
        const existing = useFiber(child, element.props)
        existing.return = returnFiber
        return existing
      } else {
        // key 相同但 type 不同，删除所有旧节点
        deleteRemainingChildren(returnFiber, child)
        break
      }
    } else {
      // key 不同，删除当前节点，继续遍历
      deleteChild(returnFiber, child)
    }
    child = child.sibling
  }
  
  // 3. 没有可复用的，创建新节点
  const created = createFiberFromElement(element, returnFiber.mode, lanes)
  created.return = returnFiber
  return created
}

// 示例
// 旧: <div key="a">A</div>
// 新: <div key="a">B</div>
// 结果：复用 div，更新文本

// 旧: <div key="a">A</div>
// 新: <span key="a">A</span>
// 结果：删除 div，创建 span（type 不同）

// 旧: <div key="a">A</div>
// 新: <div key="b">B</div>
// 结果：删除旧 div，创建新 div（key 不同）
```

### 5.3 多节点 Diff

```javascript
function reconcileChildrenArray(
  returnFiber,
  currentFirstChild,
  newChildren,
  lanes
) {
  // 多节点 Diff 分两轮遍历
  
  // ===== 第一轮：处理更新的节点 =====
  let resultingFirstChild = null
  let previousNewFiber = null
  let oldFiber = currentFirstChild
  let newIdx = 0
  
  // 遍历新旧节点，找到第一个不能复用的位置
  for (; oldFiber !== null && newIdx < newChildren.length; newIdx++) {
    const newChild = newChildren[newIdx]
    
    // key 不同，跳出第一轮遍历
    if (oldFiber.key !== newChild.key) {
      break
    }
    
    // type 相同，复用节点
    if (oldFiber.elementType === newChild.type) {
      const newFiber = useFiber(oldFiber, newChild.props)
      newFiber.index = newIdx
      newFiber.return = returnFiber
      
      if (previousNewFiber === null) {
        resultingFirstChild = newFiber
      } else {
        previousNewFiber.sibling = newFiber
      }
      previousNewFiber = newFiber
    } else {
      // type 不同，跳出第一轮遍历
      break
    }
    
    oldFiber = oldFiber.sibling
  }
  
  // 情况1：新节点遍历完，删除剩余旧节点
  if (newIdx === newChildren.length) {
    deleteRemainingChildren(returnFiber, oldFiber)
    return resultingFirstChild
  }
  
  // 情况2：旧节点遍历完，创建剩余新节点
  if (oldFiber === null) {
    for (; newIdx < newChildren.length; newIdx++) {
      const newFiber = createChild(returnFiber, newChildren[newIdx], lanes)
      if (newFiber === null) continue
      
      newFiber.index = newIdx
      if (previousNewFiber === null) {
        resultingFirstChild = newFiber
      } else {
        previousNewFiber.sibling = newFiber
      }
      previousNewFiber = newFiber
    }
    return resultingFirstChild
  }
  
  // ===== 第二轮：处理乱序节点 =====
  // 将剩余旧节点放入 Map
  const existingChildren = mapRemainingChildren(returnFiber, oldFiber)
  
  for (; newIdx < newChildren.length; newIdx++) {
    const newChild = newChildren[newIdx]
    
    // 从 Map 中查找可复用节点
    const newFiber = updateFromMap(
      existingChildren,
      returnFiber,
      newIdx,
      newChild,
      lanes
    )
    
    if (newFiber !== null) {
      // 复用节点，从 Map 中删除
      if (newFiber.alternate !== null) {
        existingChildren.delete(
          newFiber.key === null ? newIdx : newFiber.key
        )
      }
      
      // 标记移动
      placeChild(newFiber, lastPlacedIndex, newIdx)
      
      if (previousNewFiber === null) {
        resultingFirstChild = newFiber
      } else {
        previousNewFiber.sibling = newFiber
      }
      previousNewFiber = newFiber
    }
  }
  
  // 删除 Map 中剩余的旧节点
  existingChildren.forEach(child => deleteChild(returnFiber, child))
  
  return resultingFirstChild
}

// 将旧节点放入 Map
function mapRemainingChildren(returnFiber, currentFirstChild) {
  const existingChildren = new Map()
  
  let existingChild = currentFirstChild
  while (existingChild !== null) {
    const key = existingChild.key === null 
      ? existingChild.index 
      : existingChild.key
    existingChildren.set(key, existingChild)
    existingChild = existingChild.sibling
  }
  
  return existingChildren
}

// 判断是否需要移动
function placeChild(newFiber, lastPlacedIndex, newIndex) {
  newFiber.index = newIndex
  const current = newFiber.alternate
  
  if (current !== null) {
    const oldIndex = current.index
    if (oldIndex < lastPlacedIndex) {
      // 需要移动
      newFiber.flags |= Placement
      return lastPlacedIndex
    } else {
      // 不需要移动
      return oldIndex
    }
  } else {
    // 新节点，标记插入
    newFiber.flags |= Placement
    return lastPlacedIndex
  }
}
```

### 6.2 useState 实现

```javascript
// Mount 阶段
function mountState(initialState) {
  // 1. 创建 Hook
  const hook = mountWorkInProgressHook()
  
  // 2. 初始化 state
  if (typeof initialState === 'function') {
    initialState = initialState()
  }
  hook.memoizedState = hook.baseState = initialState
  
  // 3. 创建更新队列
  const queue = {
    pending: null,
    dispatch: null,
    lastRenderedReducer: basicStateReducer,
    lastRenderedState: initialState
  }
  hook.queue = queue
  
  // 4. 创建 dispatch 函数
  const dispatch = dispatchSetState.bind(
    null,
    currentlyRenderingFiber,  // 当前 Fiber
    queue
  )
  queue.dispatch = dispatch
  
  return [hook.memoizedState, dispatch]
}

// Update 阶段
function updateState(initialState) {
  return updateReducer(basicStateReducer, initialState)
}

function updateReducer(reducer, initialArg) {
  // 1. 获取当前 Hook
  const hook = updateWorkInProgressHook()
  const queue = hook.queue
  
  // 2. 处理更新队列
  const pending = queue.pending
  if (pending !== null) {
    // 计算新 state
    const first = pending.next
    let newState = hook.baseState
    
    let update = first
    do {
      const action = update.action
      newState = reducer(newState, action)
      update = update.next
    } while (update !== first)
    
    hook.memoizedState = newState
    hook.baseState = newState
    queue.pending = null
  }
  
  const dispatch = queue.dispatch
  return [hook.memoizedState, dispatch]
}

// dispatch 函数
function dispatchSetState(fiber, queue, action) {
  // 1. 创建 update 对象
  const update = {
    lane,
    action,
    next: null
  }
  
  // 2. 加入更新队列（环形链表）
  const pending = queue.pending
  if (pending === null) {
    update.next = update
  } else {
    update.next = pending.next
    pending.next = update
  }
  queue.pending = update
  
  // 3. 调度更新
  scheduleUpdateOnFiber(fiber, lane)
}

// 基础 reducer
function basicStateReducer(state, action) {
  return typeof action === 'function' ? action(state) : action
}
```

### 6.3 useEffect 实现

```javascript
// Mount 阶段
function mountEffect(create, deps) {
  return mountEffectImpl(
    PassiveEffect,  // 异步执行
    HookPassive,
    create,
    deps
  )
}

function mountEffectImpl(fiberFlags, hookFlags, create, deps) {
  // 1. 创建 Hook
  const hook = mountWorkInProgressHook()
  const nextDeps = deps === undefined ? null : deps
  
  // 2. 标记副作用
  currentlyRenderingFiber.flags |= fiberFlags
  
  // 3. 创建 effect 对象
  hook.memoizedState = pushEffect(
    HookHasEffect | hookFlags,
    create,
    undefined,  // destroy
    nextDeps
  )
}

function pushEffect(tag, create, destroy, deps) {
  const effect = {
    tag,
    create,
    destroy,
    deps,
    next: null
  }
  
  // 加入 effect 链表
  let componentUpdateQueue = currentlyRenderingFiber.updateQueue
  if (componentUpdateQueue === null) {
    componentUpdateQueue = createFunctionComponentUpdateQueue()
    currentlyRenderingFiber.updateQueue = componentUpdateQueue
    componentUpdateQueue.lastEffect = effect.next = effect
  } else {
    const lastEffect = componentUpdateQueue.lastEffect
    const firstEffect = lastEffect.next
    lastEffect.next = effect
    effect.next = firstEffect
    componentUpdateQueue.lastEffect = effect
  }
  
  return effect
}

// Update 阶段
function updateEffect(create, deps) {
  return updateEffectImpl(PassiveEffect, HookPassive, create, deps)
}

function updateEffectImpl(fiberFlags, hookFlags, create, deps) {
  const hook = updateWorkInProgressHook()
  const nextDeps = deps === undefined ? null : deps
  let destroy = undefined
  
  if (currentHook !== null) {
    const prevEffect = currentHook.memoizedState
    destroy = prevEffect.destroy
    
    if (nextDeps !== null) {
      const prevDeps = prevEffect.deps
      // 比较依赖
      if (areHookInputsEqual(nextDeps, prevDeps)) {
        // 依赖未变，不执行 effect
        hook.memoizedState = pushEffect(hookFlags, create, destroy, nextDeps)
        return
      }
    }
  }
  
  // 依赖变化，标记需要执行
  currentlyRenderingFiber.flags |= fiberFlags
  hook.memoizedState = pushEffect(
    HookHasEffect | hookFlags,
    create,
    destroy,
    nextDeps
  )
}

// Commit 阶段执行 effect
function commitPassiveEffects(finishedWork) {
  // 1. 执行上次的 destroy
  commitPassiveUnmountEffects(finishedWork)
  
  // 2. 执行本次的 create
  commitPassiveMountEffects(finishedWork)
}

function commitPassiveMountEffects(finishedWork) {
  const updateQueue = finishedWork.updateQueue
  const lastEffect = updateQueue !== null ? updateQueue.lastEffect : null
  
  if (lastEffect !== null) {
    const firstEffect = lastEffect.next
    let effect = firstEffect
    do {
      if ((effect.tag & HookHasEffect) !== NoHookEffect) {
        const create = effect.create
        // 执行 effect，保存 destroy
        effect.destroy = create()
      }
      effect = effect.next
    } while (effect !== firstEffect)
  }
}
```

### 6.4 useMemo / useCallback

```javascript
// useMemo
function mountMemo(nextCreate, deps) {
  const hook = mountWorkInProgressHook()
  const nextDeps = deps === undefined ? null : deps
  
  // 执行 create 函数，计算值
  const nextValue = nextCreate()
  
  hook.memoizedState = [nextValue, nextDeps]
  return nextValue
}

function updateMemo(nextCreate, deps) {
  const hook = updateWorkInProgressHook()
  const nextDeps = deps === undefined ? null : deps
  const prevState = hook.memoizedState
  
  if (prevState !== null && nextDeps !== null) {
    const prevDeps = prevState[1]
    // 比较依赖
    if (areHookInputsEqual(nextDeps, prevDeps)) {
      // 依赖未变，返回缓存值
      return prevState[0]
    }
  }
  
  // 依赖变化，重新计算
  const nextValue = nextCreate()
  hook.memoizedState = [nextValue, nextDeps]
  return nextValue
}

// useCallback = useMemo(() => fn)
function mountCallback(callback, deps) {
  const hook = mountWorkInProgressHook()
  const nextDeps = deps === undefined ? null : deps
  hook.memoizedState = [callback, nextDeps]
  return callback
}

function updateCallback(callback, deps) {
  const hook = updateWorkInProgressHook()
  const nextDeps = deps === undefined ? null : deps
  const prevState = hook.memoizedState
  
  if (prevState !== null && nextDeps !== null) {
    const prevDeps = prevState[1]
    if (areHookInputsEqual(nextDeps, prevDeps)) {
      return prevState[0]  // 返回旧函数
    }
  }
  
  hook.memoizedState = [callback, nextDeps]
  return callback  // 返回新函数
}
```

### Q2: React 18 的并发渲染是如何实现的？

**关键点**：

### Q5: Diff 算法的时间复杂度是多少？

**O(n)**

传统 Diff：O(n³)
- 遍历树1的每个节点：O(n)
- 遍历树2的每个节点：O(n)
- 编辑距离算法：O(n)
- 总计：O(n³)

React Diff：O(n)
- 只对同层级比较：O(n)
- 通过 key 快速定位：O(1)
- 总计：O(n)
