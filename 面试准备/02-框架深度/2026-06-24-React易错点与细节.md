## 一、super(props) 的细节

### 1.1 为什么要传递 props？

```javascript
class MyComponent extends React.Component {
  constructor(props) {
    // ❌ 不传 props
    super()
    
    // 在构造函数中访问 this.props 会是 undefined
    console.log(this.props)  // undefined ❌
  }
  
  componentDidMount() {
    // 初始化完成后，React 会自动绑定 props
    console.log(this.props)  // { name: "React" } ✅
  }
}

// ✅ 正确做法
class MyComponent extends React.Component {
  constructor(props) {
    super(props)  // 传递 props
    
    // 现在可以正常访问
    console.log(this.props)  // { name: "React" } ✅
  }
}
```

### 1.2 原理解析

```javascript
// React.Component 的简化实现
class Component {
  constructor(props) {
    this.props = props
  }
}

// 你的组件
class MyComponent extends Component {
  constructor(props) {
    // 不传 props
    super()  // 相当于 Component.call(this)
    
    // 此时 Component 的 constructor 中没有收到 props
    // 所以 this.props 没有被赋值
    console.log(this.props)  // undefined
  }
}

// React 内部会在 constructor 执行后做这个操作
instance.props = props

// 所以在 constructor 之外可以访问 this.props
```

### 1.3 最佳实践

```javascript
// ✅ 推荐：始终传递 props
class MyComponent extends React.Component {
  constructor(props) {
    super(props)
    
    // 基于 props 初始化 state
    this.state = {
      count: props.initialCount || 0
    }
  }
}

// ✅ 更推荐：使用类属性（不需要 constructor）
class MyComponent extends React.Component {
  state = {
    count: this.props.initialCount || 0
  }
}

// ✅ 最推荐：使用函数组件
function MyComponent({ initialCount = 0 }) {
  const [count, setCount] = useState(initialCount)
  return <div>{count}</div>
}
```

## 二、setState 的细节

### 2.1 异步更新的陷阱

```javascript
class Counter extends React.Component {
  state = { count: 0 }
  
  handleClick = () => {
    // ❌ 错误：连续调用不会累加
    this.setState({ count: this.state.count + 1 })
    this.setState({ count: this.state.count + 1 })
    this.setState({ count: this.state.count + 1 })
    
    // 结果：count 只增加 1，不是 3
    // 原因：this.state.count 在三次调用时都是 0
  }
  
  handleClickCorrect = () => {
    // ✅ 正确：使用函数式更新
    this.setState(prevState => ({ count: prevState.count + 1 }))
    this.setState(prevState => ({ count: prevState.count + 1 }))
    this.setState(prevState => ({ count: prevState.count + 1 }))
    
    // 结果：count 增加 3
  }
}
```

### 2.2 同步 vs 异步

```javascript
class MyComponent extends React.Component {
  state = { count: 0 }
  
  handleClick = () => {
    // 情况1：React 事件处理函数中（异步）
    this.setState({ count: 1 })
    console.log(this.state.count)  // 0（旧值）
  }
  
  componentDidMount() {
    // 情况2：生命周期中（异步）
    this.setState({ count: 1 })
    console.log(this.state.count)  // 0（旧值）
    
    // 情况3：setTimeout 中（React 17 及以前是同步，React 18 是异步）
    setTimeout(() => {
      this.setState({ count: 2 })
      console.log(this.state.count)  
      // React 17: 2（新值，同步更新）
      // React 18: 0（旧值，自动批处理）
    }, 0)
    
    // 情况4：原生事件中（React 17 及以前是同步）
    document.addEventListener('click', () => {
      this.setState({ count: 3 })
      console.log(this.state.count)
      // React 17: 3（同步更新）
      // React 18: 0（自动批处理）
    })
  }
}
```

### 2.3 setState 的回调

```javascript
class MyComponent extends React.Component {
  state = { count: 0 }
  
  handleClick = () => {
    // 使用回调获取更新后的值
    this.setState(
      { count: this.state.count + 1 },
      () => {
        // 状态已经更新完成
        console.log(this.state.count)  // 1（新值）
      }
    )
  }
}
```

## 三、事件处理的细节

### 3.1 事件对象的持久化（React 16）

```javascript
class MyComponent extends React.Component {
  handleClick = (e) => {
    // React 16 及以前：事件对象会被重置
    console.log(e.type)  // 'click'
    
    setTimeout(() => {
      console.log(e.type)  // null ❌（事件对象已被回收）
    }, 0)
  }
  
  handleClickCorrect = (e) => {
    // ✅ 方法1：持久化事件对象
    e.persist()
    
    setTimeout(() => {
      console.log(e.type)  // 'click' ✅
    }, 0)
  }
  
  handleClickCorrect2 = (e) => {
    // ✅ 方法2：提前保存需要的值
    const type = e.type
    
    setTimeout(() => {
      console.log(type)  // 'click' ✅
    }, 0)
  }
}

// React 17+ 已经移除事件池，不再需要 e.persist()
```

### 3.2 阻止事件传播的注意事项

```javascript
class MyComponent extends React.Component {
  componentDidMount() {
    // 原生事件
    document.addEventListener('click', () => {
      console.log('document click')
    })
  }
  
  handleButtonClick = (e) => {
    e.stopPropagation()  // 只阻止 React 事件冒泡
    console.log('button click')
  }
  
  render() {
    return <button onClick={this.handleButtonClick}>点击</button>
  }
}

// 点击按钮输出：
// button click
// document click  （原生事件仍会触发！）

// 如果要阻止原生事件：
handleButtonClick = (e) => {
  e.nativeEvent.stopImmediatePropagation()
}
```

### 3.3 onChange 的行为差异

```javascript
// React 的 onChange 相当于原生的 oninput

// 每次输入都会触发，而不是失去焦点时
```

## 四、Hooks 的细节

### 4.1 useEffect 的依赖数组

```javascript
function MyComponent({ id }) {
  const [data, setData] = useState(null)
  
  // ❌ 错误：依赖缺失
  useEffect(() => {
    fetchData(id).then(setData)
  }, [])  // 缺少 id 依赖
  
  // 问题：id 变化时不会重新请求数据
  
  // ✅ 正确：添加所有依赖
  useEffect(() => {
    fetchData(id).then(setData)
  }, [id])
  
  // ⚠️ 注意：函数作为依赖
  const fetchUser = () => {
    // ...
  }
  
  useEffect(() => {
    fetchUser()
  }, [fetchUser])  // ❌ 每次渲染 fetchUser 都是新函数，导致无限循环
  
  // ✅ 使用 useCallback
  const fetchUser = useCallback(() => {
    // ...
  }, [])
  
  useEffect(() => {
    fetchUser()
  }, [fetchUser])  // ✅ 正确
}
```

### 4.3 useEffect 的清理函数

```javascript
function MyComponent() {
  useEffect(() => {
    const timer = setInterval(() => {
      console.log('tick')
    }, 1000)
    
    // ❌ 忘记清理
    // 问题：组件卸载后定时器仍在运行，导致内存泄漏
  }, [])
  
  // ✅ 正确清理
  useEffect(() => {
    const timer = setInterval(() => {
      console.log('tick')

    }

    const subscription = subscribe(id, callback)

      subscription.unsubscribe()  // 取消订阅
    }
  }, [id])
  
  // 事件监听示例
  useEffect(() => {
    const handleResize = () => {
      // ...
    }
    
    window.addEventListener('resize', handleResize)
    
    return () => {
      window.removeEventListener('resize', handleResize)
    }
  }, [])
}
```

### 4.4 useRef 的多种用途

```javascript
function MyComponent() {
  // 用途1：访问 DOM
  const inputRef = useRef(null)
  
  useEffect(() => {
    inputRef.current.focus()
  }, [])
  
  // 用途2：保存可变值（不触发重渲染）
  const countRef = useRef(0)

    countRef.current++  // 不会触发重渲染

  }

  const [count, setCount] = useState(0)
  const prevCountRef = useRef()
  
  useEffect(() => {
    prevCountRef.current = count
  })
  
  const prevCount = prevCountRef.current
  
  // 用途4：避免闭包陷阱
  const [seconds, setSeconds] = useState(0)
  const secondsRef = useRef(seconds)
  
  useEffect(() => {
    secondsRef.current = seconds
  }, [seconds])
  
  useEffect(() => {
    const timer = setInterval(() => {
      // ✅ 始终访问最新的值
      console.log(secondsRef.current)
    }, 1000)
```

### 4.5 useCallback 和 useMemo 的误区

```javascript
function MyComponent({ items }) {
  // ❌ 过度优化：简单计算不需要 useMemo
  const count = useMemo(() => items.length, [items])
  // items.length 本身就很快，useMemo 反而增加开销
  
  // ✅ 有必要：复杂计算
  const expensiveValue = useMemo(() => {
    return items.reduce((sum, item) => {
      // 复杂计算...
      return sum + expensiveOperation(item)
    }, 0)

    console.log('clicked')
  }, [])
  
  return <button onClick={handleClick}>点击</button>
  // 如果 button 没有用 React.memo 包裹，useCallback 没有意义
  
  // ✅ 有意义的 useCallback
  return <ExpensiveChild onClick={handleClick} />
}

const ExpensiveChild = React.memo(({ onClick }) => {
```

## 五、key 的细节

### 5.1 index 作为 key 的问题

```javascript
function TodoList({ todos }) {
  // ❌ 错误：使用 index 作为 key
  return (
    <ul>
      {todos.map((todo, index) => (
        <li key={index}>{todo.text}</li>
      ))}
    </ul>
  )
  
  // 问题演示：
  // 初始: [{ text: 'A' }, { text: 'B' }, { text: 'C' }]
  // key:   0               1               2
  
  // 删除第一项后: [{ text: 'B' }, { text: 'C' }]
  // key:           0               1
  
  // React 认为:
  // - key=0 的项从 'A' 变成了 'B'（更新）
  // - key=1 的项从 'B' 变成了 'C'（更新）
  // - key=2 的项被删除
  
  // 实际应该:
  // - key='A' 的项被删除
  // - key='B' 和 key='C' 的项不变（复用）
  
  // ✅ 正确：使用稳定的 id
  return (
    <ul>
      {todos.map(todo => (
        <li key={todo.id}>{todo.text}</li>
      ))}
    </ul>
  )
}
```

```javascript
function List({ items }) {
```

## 八、面试易错题

### 8.1 闭包陷阱

```javascript
function Counter() {
  const [count, setCount] = useState(0)

    const timer = setInterval(() => {

  }, [])

    }, 1000)
    
    return () => clearInterval(timer)

  // ✅ 解决方案3：使用 ref
  const countRef = useRef(count)
  
  useEffect(() => {
    countRef.current = count
  }, [count])
```

### 8.2 批量更新的理解

```javascript
function Counter() {
  const [count, setCount] = useState(0)
  
  const handleClick = () => {
    console.log('start:', count)  // 0
    
    // React 18: 自动批处理
    setCount(count + 1)
    setCount(count + 1)
    setCount(count + 1)
    
    console.log('end:', count)  // 0（尚未更新）
    
    // 只会触发一次重渲染，count 变成 1（不是 3）
  }
  
  const handleClickAsync = () => {
    // React 18: 异步中也会批处理
    setTimeout(() => {
      setCount(count + 1)
      setCount(count + 1)
      setCount(count + 1)
      // 仍然只会变成 1
    }, 0)
  }
  
  const handleClickFlushSync = () => {
    // 强制同步更新
    flushSync(() => {
      setCount(count + 1)
    })
    console.log(count)  // 1（已更新）
    
    flushSync(() => {
      setCount(count + 1)
    })
    console.log(count)  // 2（已更新）
  }
  
  return <button onClick={handleClick}>{count}</button>
}
```

### 8.3 React.memo 的比较

```javascript
// ❌ React.memo 默认是浅比较
const Child = React.memo(function Child({ user }) {
  console.log('Child render')
  return <div>{user.name}</div>
})

function Parent() {
  const [count, setCount] = useState(0)
  
  const user = { name: 'React' }  // 每次渲染都是新对象
  
  return (
    <>
      <button onClick={() => setCount(count + 1)}>+</button>
      <Child user={user} />  {/* 每次都会重渲染 */}
    </>
  )
}

// ✅ 解决方案1：使用 useMemo
function Parent() {
  const [count, setCount] = useState(0)
  
  const user = useMemo(() => ({ name: 'React' }), [])
  
  return (
    <>
      <button onClick={() => setCount(count + 1)}>+</button>
      <Child user={user} />  {/* 不会重渲染 */}
    </>
  )
}

// ✅ 解决方案2：自定义比较函数
const Child = React.memo(
  function Child({ user }) {
    console.log('Child render')
    return <div>{user.name}</div>
  },
  (prevProps, nextProps) => {
    // 返回 true 表示相等（不重渲染）
    return prevProps.user.name === nextProps.user.name
  }
)
```

## 九、总结

### 核心要点

1. **super(props)**：在 constructor 中需要访问 this.props 时必须传递
2. **setState**：异步更新、批量处理、函数式更新
3. **事件系统**：合成事件、事件池（React 16）、阻止传播
4. **Hooks**：依赖数组、清理函数、闭包陷阱、性能优化
5. **key**：避免使用 index、保持稳定性
6. **渲染**：短路求值、null vs undefined、Context 性能
7. **性能优化**：React.memo、useMemo、useCallback 的正确使用
