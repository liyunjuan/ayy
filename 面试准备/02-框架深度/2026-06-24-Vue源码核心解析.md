## 二、响应式原理

### 2.1 Vue 2 响应式（Object.defineProperty）

```javascript
// Vue 2 的响应式实现
class Observer {
  constructor(value) {
    this.value = value
    
    if (Array.isArray(value)) {
      // 数组特殊处理
      this.observeArray(value)
    } else {
      // 对象处理
      this.walk(value)
    }
  }
  
  walk(obj) {
    const keys = Object.keys(obj)
    for (let i = 0; i < keys.length; i++) {
      defineReactive(obj, keys[i])
    }
  }
  
  observeArray(items) {
    for (let i = 0; i < items.length; i++) {
      observe(items[i])
    }
  }
}

// 定义响应式属性
function defineReactive(obj, key, val) {
  // 递归观察子属性
  let childOb = observe(val)
  
  // 每个属性都有一个 Dep 实例
  const dep = new Dep()
  
  Object.defineProperty(obj, key, {
    enumerable: true,
    configurable: true,
    
    get() {
      // 依赖收集
      if (Dep.target) {
        dep.depend()
        
        // 子对象也要收集依赖
        if (childOb) {
          childOb.dep.depend()
        }
      }
      return val
    },
    
    set(newVal) {
      if (newVal === val) return
      
      val = newVal
      // 新值也要观察
      childOb = observe(newVal)
      
      // 派发更新
      dep.notify()
    }
  })
}

// 依赖收集器
class Dep {
  constructor() {
    this.subs = []  // 存储 Watcher
  }
  
  addSub(sub) {
    this.subs.push(sub)
  }
  
  depend() {
    if (Dep.target) {
      Dep.target.addDep(this)
    }
  }
  
  notify() {
    // 通知所有 Watcher 更新
    const subs = this.subs.slice()
    for (let i = 0; i < subs.length; i++) {
      subs[i].update()
    }
  }
}

Dep.target = null  // 当前正在计算的 Watcher

// 观察者
class Watcher {
  constructor(vm, expOrFn, cb) {
    this.vm = vm
    this.cb = cb
    this.deps = []
    this.depIds = new Set()
    
    // 支持函数或表达式
    if (typeof expOrFn === 'function') {
      this.getter = expOrFn
    } else {
      this.getter = parsePath(expOrFn)
    }
    
    this.value = this.get()
  }
  
  get() {
    // 设置当前 Watcher
    Dep.target = this
    
    // 触发 getter，收集依赖
    const value = this.getter.call(this.vm, this.vm)
    
    // 清空当前 Watcher
    Dep.target = null
    
    return value
  }
  
  addDep(dep) {
    const id = dep.id
    if (!this.depIds.has(id)) {
      this.depIds.add(id)
      this.deps.push(dep)
      dep.addSub(this)
    }
  }
  
  update() {
    // 触发更新
    const oldValue = this.value
    this.value = this.get()
    this.cb.call(this.vm, this.value, oldValue)
  }
}

// 使用示例
const data = {
  message: 'Hello',
  user: {
    name: 'Vue'
  }
}

observe(data)

// 创建 Watcher
new Watcher(data, () => {
  console.log('message changed:', data.message)
}, () => {})

data.message = 'World'  // 触发更新
```

### 2.2 Vue 3 响应式（Proxy）

```javascript
// packages/reactivity/src/reactive.ts

// 响应式 Map
const reactiveMap = new WeakMap()

// 创建响应式对象
function reactive(target) {
  // 如果不是对象，直接返回
  if (typeof target !== 'object' || target === null) {
    return target
  }
  
  // 如果已经是响应式对象，直接返回
  if (reactiveMap.has(target)) {
    return reactiveMap.get(target)
  }
  
  // 创建 Proxy
  const proxy = new Proxy(target, {
    get(target, key, receiver) {
      // 特殊 key 处理
      if (key === '__v_isReactive') {
        return true
      }
      
      // 依赖收集
      track(target, key)
      
      const res = Reflect.get(target, key, receiver)
      
      // 如果是对象，递归代理（懒代理）
      if (typeof res === 'object' && res !== null) {
        return reactive(res)
      }
      
      return res
    },
    
    set(target, key, value, receiver) {
      const oldValue = target[key]
      const result = Reflect.set(target, key, value, receiver)
      
      // 值发生变化，触发更新
      if (oldValue !== value) {
        trigger(target, key)
      }
      
      return result
    },
    
    deleteProperty(target, key) {
      const hadKey = Object.prototype.hasOwnProperty.call(target, key)
      const result = Reflect.deleteProperty(target, key)
      
      if (hadKey && result) {
        trigger(target, key)
      }
      
      return result
    },
    
    has(target, key) {
      track(target, key)
      return Reflect.has(target, key)
    },
    
    ownKeys(target) {
      track(target, Array.isArray(target) ? 'length' : Symbol('iterate'))
      return Reflect.ownKeys(target)
    }
  })
  
  reactiveMap.set(target, proxy)
  return proxy
}

// 依赖收集的数据结构
// WeakMap<target, Map<key, Set<effect>>>
const targetMap = new WeakMap()

// 当前活动的 effect
let activeEffect = null
const effectStack = []

// 依赖收集
function track(target, key) {
  if (!activeEffect) return
  
  let depsMap = targetMap.get(target)
  if (!depsMap) {
    targetMap.set(target, (depsMap = new Map()))
  }
  
  let dep = depsMap.get(key)
  if (!dep) {
    depsMap.set(key, (dep = new Set()))
  }
  
  // 双向记录：effect 记录 dep，dep 记录 effect
  if (!dep.has(activeEffect)) {
    dep.add(activeEffect)
    activeEffect.deps.push(dep)
  }
}

// 派发更新
function trigger(target, key) {
  const depsMap = targetMap.get(target)
  if (!depsMap) return
  
  const effects = depsMap.get(key)
  if (!effects) return
  
  // 避免无限循环
  const effectsToRun = new Set(effects)
  
  effectsToRun.forEach(effect => {
    // 如果有调度器，使用调度器
    if (effect.scheduler) {
      effect.scheduler()
    } else {
      effect()
    }
  })
}

// 副作用函数
function effect(fn, options = {}) {
  const effectFn = () => {
    try {
      // 清理旧的依赖
      cleanup(effectFn)
      
      // 设置当前 effect
      activeEffect = effectFn
      effectStack.push(effectFn)
      
      // 执行函数，触发依赖收集
      return fn()
    } finally {
      effectStack.pop()
      activeEffect = effectStack[effectStack.length - 1]
    }
  }
  
  effectFn.deps = []
  effectFn.options = options
  
  if (options.scheduler) {
    effectFn.scheduler = options.scheduler
  }
  
  // 立即执行
  if (!options.lazy) {
    effectFn()
  }
  
  return effectFn
}

// 清理依赖
function cleanup(effect) {
  const { deps } = effect
  if (deps.length) {
    for (let i = 0; i < deps.length; i++) {
      deps[i].delete(effect)
    }
    deps.length = 0
  }
}

// 使用示例
const state = reactive({
  count: 0,
  user: {
    name: 'Vue'
  }
})

// 创建副作用
effect(() => {
  console.log('count changed:', state.count)
})

state.count++  // 输出: count changed: 1

// ✅ Vue 3 解决了 Vue 2 的问题
state.newProp = 'new'  // ✅ 可以检测
delete state.count  // ✅ 可以检测

const arr = reactive([1, 2, 3])
arr[0] = 4  // ✅ 可以检测
arr.length = 0  // ✅ 可以检测
```

### 2.3 ref 实现

```javascript
// packages/reactivity/src/ref.ts

class RefImpl {
  constructor(value) {
    this._value = convert(value)  // 如果是对象，转为 reactive
    this.__v_isRef = true
  }
  
  get value() {
    // 依赖收集
    track(this, 'value')
    return this._value
  }
  
  set value(newValue) {
    if (newValue !== this._value) {
      this._value = convert(newValue)
      // 派发更新
      trigger(this, 'value')
    }
  }
}

function convert(value) {
  return typeof value === 'object' ? reactive(value) : value
}

function ref(value) {
  return new RefImpl(value)
}

// 自动解包
function proxyRefs(objectWithRefs) {
  return new Proxy(objectWithRefs, {
    get(target, key, receiver) {
      const value = Reflect.get(target, key, receiver)
      // 如果是 ref，自动解包
      return value.__v_isRef ? value.value : value
    },
    
    set(target, key, value, receiver) {
      const oldValue = target[key]
      // 如果旧值是 ref，新值不是，则赋值给 .value
      if (oldValue.__v_isRef && !value.__v_isRef) {
        oldValue.value = value
        return true
      }
      return Reflect.set(target, key, value, receiver)
    }
  })
}

// 使用示例
const count = ref(0)

effect(() => {
  console.log('count:', count.value)
})

count.value++  // 输出: count: 1

// 在 setup 返回的对象中自动解包
const setupResult = proxyRefs({
  count,
  message: ref('Hello')
})

console.log(setupResult.count)  // 0（自动解包，不需要 .value）
```

### 2.4 computed 实现

```javascript
// packages/reactivity/src/computed.ts

class ComputedRefImpl {
  constructor(getter, setter) {
    this._getter = getter
    this._setter = setter
    this._dirty = true  // 脏标记
    this._value = undefined
    
    // 创建 effect，但不立即执行
    this.effect = effect(getter, {
      lazy: true,
      scheduler: () => {
        // 依赖变化时，标记为脏
        if (!this._dirty) {
          this._dirty = true
          // 触发依赖这个 computed 的 effect
          trigger(this, 'value')
        }
      }
    })
  }
  
  get value() {
    // 依赖收集
    track(this, 'value')
    
    // 只有脏了才重新计算
    if (this._dirty) {
      this._value = this.effect()
      this._dirty = false
    }
    
    return this._value
  }
  
  set value(newValue) {
    this._setter(newValue)
  }
}

function computed(getterOrOptions) {
  let getter, setter
  
  if (typeof getterOrOptions === 'function') {
    getter = getterOrOptions
    setter = () => {
      console.warn('Computed property is readonly')
    }
  } else {
    getter = getterOrOptions.get
    setter = getterOrOptions.set
  }
  
  return new ComputedRefImpl(getter, setter)
}

// 使用示例
const state = reactive({ count: 0 })

const double = computed(() => {
  console.log('computing...')
  return state.count * 2
})

console.log(double.value)  // 输出: computing... 0
console.log(double.value)  // 0（不会重新计算，因为依赖没变）

state.count++
console.log(double.value)  // 输出: computing... 2（重新计算）
```

```javascript
// packages/runtime-core/src/vnode.ts

// VNode 类型
export const Fragment = Symbol('Fragment')
export const Text = Symbol('Text')
export const Comment = Symbol('Comment')

function createVNode(type, props = null, children = null) {
  const vnode = {
    type,           // 组件类型或 HTML 标签
    props,          // 属性
    children,       // 子节点
    key: props?.key,
    el: null,       // 真实 DOM 节点
    component: null, // 组件实例
    shapeFlag: 0,   // 节点类型标记
    patchFlag: 0,   // 优化标记
  }
  
  // 确定 shapeFlag
  if (typeof type === 'string') {
    vnode.shapeFlag = ShapeFlags.ELEMENT
  } else if (typeof type === 'object') {
    vnode.shapeFlag = ShapeFlags.STATEFUL_COMPONENT
  } else if (typeof type === 'function') {
    vnode.shapeFlag = ShapeFlags.FUNCTIONAL_COMPONENT
  }
  
  // 确定子节点类型
  if (typeof children === 'string') {
    vnode.shapeFlag |= ShapeFlags.TEXT_CHILDREN
  } else if (Array.isArray(children)) {
    vnode.shapeFlag |= ShapeFlags.ARRAY_CHILDREN
  }
  
  return vnode
}

// ShapeFlags：位运算标记
export const enum ShapeFlags {
  ELEMENT = 1,                    // 1
  FUNCTIONAL_COMPONENT = 1 << 1,  // 2
  STATEFUL_COMPONENT = 1 << 2,    // 4
  TEXT_CHILDREN = 1 << 3,         // 8
  ARRAY_CHILDREN = 1 << 4,        // 16
  SLOTS_CHILDREN = 1 << 5,        // 32
  COMPONENT = ShapeFlags.STATEFUL_COMPONENT | ShapeFlags.FUNCTIONAL_COMPONENT  // 6
}

// 判断节点类型
const isElement = vnode.shapeFlag & ShapeFlags.ELEMENT
const isComponent = vnode.shapeFlag & ShapeFlags.COMPONENT
```

### 3.2 渲染器（Renderer）

```javascript
// packages/runtime-core/src/renderer.ts

function createRenderer(options) {
  const {
    createElement,
    setElementText,
    patchProp,
    insert,
    remove
  } = options
  
  // 挂载元素
  function mountElement(vnode, container) {
    // 1. 创建 DOM 元素
    const el = vnode.el = createElement(vnode.type)
    
    // 2. 处理子节点
    if (vnode.shapeFlag & ShapeFlags.TEXT_CHILDREN) {
      // 文本子节点
      setElementText(el, vnode.children)
    } else if (vnode.shapeFlag & ShapeFlags.ARRAY_CHILDREN) {
      // 数组子节点
      vnode.children.forEach(child => {
        patch(null, child, el)
      })
    }
    
    // 3. 处理 props
    if (vnode.props) {
      for (const key in vnode.props) {
        patchProp(el, key, null, vnode.props[key])
      }
    }
    
    // 4. 插入到容器
    insert(el, container)
  }
  
  // 更新元素
  function patchElement(n1, n2) {
    const el = n2.el = n1.el
    const oldProps = n1.props || {}
    const newProps = n2.props || {}
    
    // 1. 更新 props
    // 更新新 props
    for (const key in newProps) {
      if (newProps[key] !== oldProps[key]) {
        patchProp(el, key, oldProps[key], newProps[key])
      }
    }
    // 删除旧 props
    for (const key in oldProps) {
      if (!(key in newProps)) {
        patchProp(el, key, oldProps[key], null)
      }
    }
    
    // 2. 更新子节点
    patchChildren(n1, n2, el)
  }
  
  // 更新子节点
  function patchChildren(n1, n2, container) {
    const c1 = n1.children
    const c2 = n2.children
    const prevShapeFlag = n1.shapeFlag
    const shapeFlag = n2.shapeFlag
    
    // 新子节点是文本
    if (shapeFlag & ShapeFlags.TEXT_CHILDREN) {
      // 旧子节点是数组，卸载
      if (prevShapeFlag & ShapeFlags.ARRAY_CHILDREN) {
        unmountChildren(c1)
      }
      // 设置新文本
      if (c2 !== c1) {
        setElementText(container, c2)
      }
    }
    // 新子节点是数组
    else if (shapeFlag & ShapeFlags.ARRAY_CHILDREN) {
      // 旧子节点也是数组 -> Diff
      if (prevShapeFlag & ShapeFlags.ARRAY_CHILDREN) {
        patchKeyedChildren(c1, c2, container)
      }
      // 旧子节点是文本 -> 清空文本，挂载新数组
      else {
        setElementText(container, '')
        c2.forEach(child => patch(null, child, container))
      }
    }
    // 新子节点为空
    else {
      // 旧子节点是数组，卸载
      if (prevShapeFlag & ShapeFlags.ARRAY_CHILDREN) {
        unmountChildren(c1)
      }
      // 旧子节点是文本，清空
      else if (prevShapeFlag & ShapeFlags.TEXT_CHILDREN) {
        setElementText(container, '')
      }
    }
  }
  
  // 核心 patch 函数
  function patch(n1, n2, container) {
    // n1 为 null，说明是挂载
    if (n1 == null) {
      if (n2.shapeFlag & ShapeFlags.ELEMENT) {
        mountElement(n2, container)
      } else if (n2.shapeFlag & ShapeFlags.COMPONENT) {
        mountComponent(n2, container)
      }
    }
    // n1 存在，是更新
    else {
      // type 不同，直接替换
      if (n1.type !== n2.type) {
        unmount(n1)
        patch(null, n2, container)
      }
      // type 相同，进行 patch
      else {
        if (n2.shapeFlag & ShapeFlags.ELEMENT) {
          patchElement(n1, n2)
        } else if (n2.shapeFlag & ShapeFlags.COMPONENT) {
          patchComponent(n1, n2)
        }
      }
    }
  }
  
  // 渲染函数
  function render(vnode, container) {
    if (vnode == null) {
      // 卸载
      if (container._vnode) {
        unmount(container._vnode)
      }
    } else {
      // 挂载或更新
      patch(container._vnode || null, vnode, container)
    }
    container._vnode = vnode
  }
  
  return {
    render
  }
}

// 浏览器平台的 DOM 操作
const renderer = createRenderer({
  createElement(tag) {
    return document.createElement(tag)
  },
  
  setElementText(el, text) {
    el.textContent = text
  },
  
  patchProp(el, key, prevValue, nextValue) {
    if (key === 'class') {
      el.className = nextValue || ''
    } else if (key === 'style') {
      // 处理 style
    } else if (/^on[A-Z]/.test(key)) {
      // 事件
      const eventName = key.slice(2).toLowerCase()
      if (prevValue) {
        el.removeEventListener(eventName, prevValue)
      }
      if (nextValue) {
        el.addEventListener(eventName, nextValue)
      }
    } else {
      // 普通属性
      if (nextValue == null) {
        el.removeAttribute(key)
      } else {
        el.setAttribute(key, nextValue)
      }
    }
  },
  
  insert(el, parent, anchor = null) {
    parent.insertBefore(el, anchor)
  },
  
  remove(el) {
    const parent = el.parentNode
    if (parent) {
      parent.removeChild(el)
    }
  }
})
```

## 四、Diff 算法

### 4.1 简单 Diff（Vue 2）

```javascript
// Vue 2 的 Diff 算法：双端比较

function updateChildren(parentElm, oldCh, newCh) {
  let oldStartIdx = 0
  let oldEndIdx = oldCh.length - 1
  let newStartIdx = 0
  let newEndIdx = newCh.length - 1
  
  let oldStartVnode = oldCh[0]
  let oldEndVnode = oldCh[oldEndIdx]
  let newStartVnode = newCh[0]
  let newEndVnode = newCh[newEndIdx]
  
  while (oldStartIdx <= oldEndIdx && newStartIdx <= newEndIdx) {
    // 跳过已处理的节点
    if (!oldStartVnode) {
      oldStartVnode = oldCh[++oldStartIdx]
    } else if (!oldEndVnode) {
      oldEndVnode = oldCh[--oldEndIdx]
    }
    
    // 1. 旧头 vs 新头
    else if (sameVnode(oldStartVnode, newStartVnode)) {
      patchVnode(oldStartVnode, newStartVnode)
      oldStartVnode = oldCh[++oldStartIdx]
      newStartVnode = newCh[++newStartIdx]
    }
    
    // 2. 旧尾 vs 新尾
    else if (sameVnode(oldEndVnode, newEndVnode)) {
      patchVnode(oldEndVnode, newEndVnode)
      oldEndVnode = oldCh[--oldEndIdx]
      newEndVnode = newCh[--newEndIdx]
    }
    
    // 3. 旧头 vs 新尾（节点右移）
    else if (sameVnode(oldStartVnode, newEndVnode)) {
      patchVnode(oldStartVnode, newEndVnode)
      insertBefore(parentElm, oldStartVnode.el, oldEndVnode.el.nextSibling)
      oldStartVnode = oldCh[++oldStartIdx]
      newEndVnode = newCh[--newEndIdx]
    }
    
    // 4. 旧尾 vs 新头（节点左移）
    else if (sameVnode(oldEndVnode, newStartVnode)) {
      patchVnode(oldEndVnode, newStartVnode)
      insertBefore(parentElm, oldEndVnode.el, oldStartVnode.el)
      oldEndVnode = oldCh[--oldEndIdx]
      newStartVnode = newCh[++newStartIdx]
    }
    
    // 5. 以上都不满足，查找 key
    else {
      // 创建旧节点的 key -> index 映射
      const oldKeyToIdx = createKeyToOldIdx(oldCh, oldStartIdx, oldEndIdx)
      const idxInOld = oldKeyToIdx[newStartVnode.key]
      
      if (!idxInOld) {
        // 新节点，创建
        createElm(newStartVnode, parentElm, oldStartVnode.el)
      } else {
        // 复用旧节点
        const vnodeToMove = oldCh[idxInOld]
        patchVnode(vnodeToMove, newStartVnode)
        oldCh[idxInOld] = undefined
        insertBefore(parentElm, vnodeToMove.el, oldStartVnode.el)
      }
      
      newStartVnode = newCh[++newStartIdx]
    }
  }
  
  // 添加剩余新节点
  if (oldStartIdx > oldEndIdx) {
    for (; newStartIdx <= newEndIdx; newStartIdx++) {
      createElm(newCh[newStartIdx], parentElm, oldStartVnode?.el)
    }
  }
  
  // 删除剩余旧节点
  else if (newStartIdx > newEndIdx) {
    for (; oldStartIdx <= oldEndIdx; oldStartIdx++) {
      if (oldCh[oldStartIdx]) {
        removeNode(oldCh[oldStartIdx].el)
      }
    }
  }
}

function sameVnode(a, b) {
  return a.key === b.key && a.type === b.type
}
```

### 4.2 快速 Diff（Vue 3）

```javascript
// Vue 3 的 Diff 算法：优化性能

function patchKeyedChildren(c1, c2, container) {
  let i = 0
  const l2 = c2.length
  let e1 = c1.length - 1  // 旧子节点的最后一个索引
  let e2 = l2 - 1          // 新子节点的最后一个索引
  
  // 1. 从头开始同步
  // (a b) c
  // (a b) d e
  while (i <= e1 && i <= e2) {
    const n1 = c1[i]
    const n2 = c2[i]
    
    if (isSameVNodeType(n1, n2)) {
      patch(n1, n2, container)
    } else {
      break
    }
    i++
  }
  
  // 2. 从尾开始同步
  // a (b c)
  // d e (b c)
  while (i <= e1 && i <= e2) {
    const n1 = c1[e1]
    const n2 = c2[e2]
    
    if (isSameVNodeType(n1, n2)) {
      patch(n1, n2, container)
    } else {
      break
    }
    e1--
    e2--
  }
  
  // 3. 处理剩余节点
  
  // 3.1 旧节点已处理完，新节点有剩余 -> 挂载新节点
  // (a b)
  // (a b) c d
  if (i > e1) {
    if (i <= e2) {
      const nextPos = e2 + 1
      const anchor = nextPos < l2 ? c2[nextPos].el : null
      while (i <= e2) {
        patch(null, c2[i], container, anchor)
        i++
      }
    }
  }
  
  // 3.2 新节点已处理完，旧节点有剩余 -> 卸载旧节点
  // (a b) c d
  // (a b)
  else if (i > e2) {
    while (i <= e1) {
      unmount(c1[i])
      i++
    }
  }
  
  // 3.3 乱序情况
  // a [b c d] e
  // a [d c b f] e
  else {
    const s1 = i  // 旧子节点的起始索引
    const s2 = i  // 新子节点的起始索引
    
    // 3.3.1 构建新子节点的 key -> index 映射
    const keyToNewIndexMap = new Map()
    for (i = s2; i <= e2; i++) {
      const nextChild = c2[i]
      if (nextChild.key != null) {
        keyToNewIndexMap.set(nextChild.key, i)
      }
    }
    
    // 3.3.2 遍历旧子节点
    let patched = 0  // 已处理的新节点数量
    const toBePatched = e2 - s2 + 1  // 需要处理的新节点数量
    let moved = false  // 是否需要移动
    let maxNewIndexSoFar = 0
    
    // 新节点索引 -> 旧节点索引的映射
    const newIndexToOldIndexMap = new Array(toBePatched).fill(0)
    
    for (i = s1; i <= e1; i++) {
      const prevChild = c1[i]
      
      // 所有新节点都已处理，剩余的旧节点删除
      if (patched >= toBePatched) {
        unmount(prevChild)
        continue
      }
      
      let newIndex
      // 在新子节点中查找旧子节点
      if (prevChild.key != null) {
        newIndex = keyToNewIndexMap.get(prevChild.key)
      } else {
        // 没有 key，遍历查找
        for (let j = s2; j <= e2; j++) {
          if (newIndexToOldIndexMap[j - s2] === 0 && 
              isSameVNodeType(prevChild, c2[j])) {
            newIndex = j
            break
          }
        }
      }
      
      // 没找到，删除
      if (newIndex === undefined) {
        unmount(prevChild)
      }
      // 找到了，patch
      else {
        // +1 是因为 0 表示没有对应的旧节点
        newIndexToOldIndexMap[newIndex - s2] = i + 1
        
        // 判断是否需要移动
        if (newIndex >= maxNewIndexSoFar) {
          maxNewIndexSoFar = newIndex
        } else {
          moved = true
        }
        
        patch(prevChild, c2[newIndex], container)
        patched++
      }
    }
    
    // 3.3.3 移动和挂载新节点
    // 求最长递增子序列
    const increasingNewIndexSequence = moved
      ? getSequence(newIndexToOldIndexMap)
      : []
    
    let j = increasingNewIndexSequence.length - 1
    
    // 倒序遍历，方便插入
    for (i = toBePatched - 1; i >= 0; i--) {
      const nextIndex = s2 + i
      const nextChild = c2[nextIndex]
      const anchor = nextIndex + 1 < l2 ? c2[nextIndex + 1].el : null
      
      // 新节点，挂载
      if (newIndexToOldIndexMap[i] === 0) {
        patch(null, nextChild, container, anchor)
      }
      // 需要移动
      else if (moved) {
        // 不在最长递增子序列中，需要移动
        if (j < 0 || i !== increasingNewIndexSequence[j]) {
          move(nextChild, container, anchor)
        } else {
          j--
        }
      }
    }
  }
}

// 最长递增子序列（LIS）
function getSequence(arr) {
  const len = arr.length
  const result = [0]
  const p = arr.slice()
  
  for (let i = 0; i < len; i++) {
    const arrI = arr[i]
    if (arrI !== 0) {
      const j = result[result.length - 1]
      if (arr[j] < arrI) {
        p[i] = j
        result.push(i)
        continue
      }
      
      // 二分查找
      let left = 0
      let right = result.length - 1
      while (left < right) {
        const mid = (left + right) >> 1
        if (arr[result[mid]] < arrI) {
          left = mid + 1
        } else {
          right = mid
        }
      }
      
      if (arrI < arr[result[left]]) {
        if (left > 0) {
          p[i] = result[left - 1]
        }
        result[left] = i
      }
    }
  }
  
  let len2 = result.length
  let idx = result[len2 - 1]
  while (len2-- > 0) {
    result[len2] = idx
    idx = p[idx]
  }
  
  return result
}
```

| 优化 | 4 次对比 | 预处理 + LIS |

```javascript
function parseChildren(context, ancestors) {
  const nodes = []
  
  while (!isEnd(context, ancestors)) {
    let node
    const s = context.source
    
    // 解析插值 {{ }}
    if (s.startsWith('{{')) {
      node = parseInterpolation(context)
    }
    // 解析标签
    else if (s[0] === '<') {
      if (s[1] === '/') {
        // 结束标签
        continue
      } else if (/[a-z]/i.test(s[1])) {
        // 开始标签
        node = parseElement(context, ancestors)
      }
    }
    
    // 文本节点
    if (!node) {
      node = parseText(context)
    }
    
    nodes.push(node)
  }
  
  return nodes
}

// 解析元素
function parseElement(context, ancestors) {
  // 1. 解析开始标签
  const element = parseTag(context, TagType.Start)
  
  // 自闭合标签，直接返回
  if (element.isSelfClosing) {
    return element
  }
  
  // 2. 递归解析子节点
  ancestors.push(element)
  element.children = parseChildren(context, ancestors)
  ancestors.pop()
  
  // 3. 解析结束标签
  parseTag(context, TagType.End)
  
  return element
}

// 解析标签
function parseTag(context, type) {
  // <div id="app">
  const match = /^<\/?([a-z][^\s/>]*)/i.exec(context.source)
  const tag = match[1]
  
  advanceBy(context, match[0].length)
  advanceSpaces(context)
  
  // 解析属性
  let props = []
  if (type === TagType.Start) {
    props = parseAttributes(context)
  }
  
  // 自闭合？
  const isSelfClosing = context.source.startsWith('/>')
  advanceBy(context, isSelfClosing ? 2 : 1)
  
  return {
    type: NodeTypes.ELEMENT,
    tag,
    props,
    children: [],
    isSelfClosing
  }
}

// 解析插值
function parseInterpolation(context) {
  // {{ message }}
  const open = '{{'
  const close = '}}'
  
  advanceBy(context, open.length)
  
  const closeIndex = context.source.indexOf(close)
  const content = context.source.slice(0, closeIndex).trim()
  
  advanceBy(context, content.length + close.length)
  
  return {
    type: NodeTypes.INTERPOLATION,
    content: {
      type: NodeTypes.SIMPLE_EXPRESSION,
      content
    }
  }
}
```

```javascript
// packages/compiler-core/src/transform.ts

function transform(ast, options) {
  const context = createTransformContext(ast, options)
  
  // 遍历 AST
  traverseNode(ast, context)
  
  // 生成代码生成所需的信息
  createRootCodegen(ast)
  
  return ast
}

function traverseNode(node, context) {
  // 执行转换插件
  const { nodeTransforms } = context
  const exitFns = []
  
  for (let i = 0; i < nodeTransforms.length; i++) {
    const onExit = nodeTransforms[i](node, context)
    if (onExit) {
      exitFns.push(onExit)
    }
  }
  
  // 递归子节点
  switch (node.type) {
    case NodeTypes.ELEMENT:
    case NodeTypes.ROOT:
      traverseChildren(node, context)
      break
    case NodeTypes.INTERPOLATION:
      // 插值表达式
      break
  }
  
  // 执行退出函数（后序遍历）
  let i = exitFns.length
  while (i--) {
    exitFns[i]()
  }
}

// 转换插件：v-if
function transformIf(node, context) {
  if (node.type === NodeTypes.ELEMENT) {
    const ifProp = findProp(node, 'v-if')
    if (ifProp) {
      // 转换为条件节点
      return () => {
        node.codegenNode = createConditionalExpression(
          ifProp.exp,
          node,
          createCallExpression('createCommentVNode', ['"v-if"'])
        )
      }
    }
  }
}

// 转换插件：v-for
function transformFor(node, context) {
  if (node.type === NodeTypes.ELEMENT) {
    const forProp = findProp(node, 'v-for')
    if (forProp) {
      // 解析 v-for 表达式
      // item in list -> { source: 'list', value: 'item' }
      const { source, value } = parseForExpression(forProp.exp)
      
      return () => {
        node.codegenNode = createForLoop(source, value, node)
      }
    }
  }
}
```

```javascript
function setupRenderEffect(instance, container) {
  // 创建响应式副作用
  instance.update = effect(() => {
    if (!instance.isMounted) {
      // ===== 挂载 =====
      
      // 调用 beforeMount
      if (instance.bm) {
        invokeArrayFns(instance.bm)
      }
      
      // 执行 render，生成 VNode 树
      const subTree = instance.subTree = instance.render.call(
        instance.ctx,
        instance.ctx
      )
      
      // patch VNode
      patch(null, subTree, container)
      
      // 保存根 DOM
      instance.vnode.el = subTree.el
      
      // 调用 mounted
      if (instance.m) {
        queuePostFlushCb(instance.m)
      }
      
      instance.isMounted = true
    } else {
      // ===== 更新 =====
      
      // 调用 beforeUpdate
      if (instance.bu) {
        invokeArrayFns(instance.bu)
      }
      
      const nextTree = instance.render.call(instance.ctx, instance.ctx)
      const prevTree = instance.subTree
      instance.subTree = nextTree
      
      // patch
      patch(prevTree, nextTree, container)
      
      // 调用 updated
      if (instance.u) {
        queuePostFlushCb(instance.u)
      }
    }
  }, {
    scheduler: queueJob  // 使用调度器
  })
}
```

```javascript
function createHook(lifecycle) {
  return function(hook, target = currentInstance) {
    if (target) {
      // 多个相同钩子会合并成数组
      const hooks = target[lifecycle] || (target[lifecycle] = [])
      
      // 包装 hook，注入错误处理

          return hook(...args)
        } catch (err) {
          handleError(err, target, lifecycle)
        }
      }
      
      hooks.push(wrappedHook)
    }
  }
```

## 八、面试高频问题

### Q1: Vue 2 和 Vue 3 响应式的区别？

**Vue 2（Object.defineProperty）**：
```javascript
// 优点：
// - 兼容性好（IE9+）

// 缺点：
// - 无法检测对象属性的添加/删除
// - 无法检测数组索引和长度的变化
// - 需要递归遍历，性能开销大
// - 需要特殊处理数组方法

const data = { count: 0 }
Object.defineProperty(data, 'count', {
  get() { /* 收集依赖 */ },
  set() { /* 派发更新 */ }
})

// ❌ 无法检测
data.newProp = 1
delete data.count
```

**Vue 3（Proxy）**：
```javascript
// 优点：
// - 可以检测所有操作（增删改查）
// - 懒代理，性能更好
// - 不需要特殊处理数组
// - 可以代理整个对象

// 缺点：
// - 不支持 IE（需要 polyfill 但性能差）

const data = new Proxy({ count: 0 }, {
  get(target, key) { /* 收集依赖 */ },
  set(target, key, value) { /* 派发更新 */ },
  deleteProperty(target, key) { /* 派发更新 */ }
})

// ✅ 都能检测
data.newProp = 1
delete data.count
```

### Q2: Vue 的 Diff 算法原理？

**核心思想**：
1. **同层比较**：只比较同一层级，不跨级比较
2. **key 优化**：通过 key 快速定位可复用节点
3. **双端比较**（Vue 2）：从两端向中间比较
4. **快速 Diff**（Vue 3）：预处理 + 最长递增子序列

**Vue 3 的优化**：
```javascript
// 旧: a b c d e
// 新: a b f g d e

// 1. 预处理：从头同步
// a b 相同，跳过

// 2. 预处理：从尾同步
// d e 相同，跳过

// 3. 处理中间乱序部分
// c -> f g
// 只需要删除 c，插入 f g

// 4. 使用最长递增子序列减少移动
// 找到不需要移动的节点，只移动必须移动的
```

### Q3: Vue 的模板是如何编译的？

**三个阶段**：

1. **Parse（解析）**：template → AST
   ```javascript
   '<div>{{ msg }}</div>' 
   → 
   { type: 'Element', tag: 'div', children: [...] }
   ```

2. **Transform（转换）**：AST → 优化后的 AST
   ```javascript
   // 静态提升、事件缓存等优化
   // 处理指令：v-if、v-for、v-model等
   ```

3. **Generate（生成）**：AST → render 函数
   ```javascript
   → 
   function render() {
     return _createVNode('div', null, _toDisplayString(msg))
   }
   ```

### Q4: computed 和 watch 的区别？

| 特性 | computed | watch |
|------|----------|-------|
| 用途 | 计算派生值 | 监听数据变化 |
| 返回值 | 必须返回 | 不需要返回 |
| 缓存 | ✅ 有缓存 | ❌ 无缓存 |
| 异步 | ❌ 不支持 | ✅ 支持 |
| 多个依赖 | ✅ 自动收集 | ❌ 只监听一个 |

```javascript
// computed: 计算属性
const fullName = computed(() => {
  return firstName.value + ' ' + lastName.value
})

// watch: 侦听器
watch(count, (newVal, oldVal) => {
  console.log('count changed:', newVal)
})

// watchEffect: 自动收集依赖
watchEffect(() => {
  console.log('count:', count.value)
})
```

### Q5: Vue 3 的 Composition API 相比 Options API 的优势？

**Options API 的问题**：
```javascript
export default {
  data() {
    return { count: 0, user: {} }
  },
  computed: {
    double() { return this.count * 2 }
  },
  methods: {
    increment() { this.count++ }
  },
  mounted() {
    // 同一功能的代码分散在不同选项
  }
}
```

**Composition API 的优势**：
```javascript
export default {
  setup() {
    // 1. 逻辑组合：相关代码组织在一起
    const { count, double, increment } = useCounter()
    
    // 2. 代码复用：通过 composable 函数
    const { user, loading } = useUser()
    
    // 3. TypeScript 支持更好
    // 4. Tree-shaking 更友好
    
    return { count, double, increment, user, loading }
  }
}

// 可复用逻辑
function useCounter() {
  const count = ref(0)
  const double = computed(() => count.value * 2)
  const increment = () => count.value++
  
  return { count, double, increment }
}
```
