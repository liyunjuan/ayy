# Object.assign 核心要点

> 精简版：只讲最重要的部分

## 一、基本用法

```javascript
// 语法
Object.assign(target, ...sources)

// 作用：将源对象的可枚举属性复制到目标对象
const target = { a: 1 }
const source = { b: 2, c: 3 }
const result = Object.assign(target, source)

console.log(result)  // { a: 1, b: 2, c: 3 }
console.log(target)  // { a: 1, b: 2, c: 3 }（target 被修改了！）
console.log(result === target)  // true（返回的就是 target）
```

## 二、核心特性

### 2.1 浅拷贝（最重要！）

```javascript
// ❌ 嵌套对象不会深拷贝
const obj1 = {
  a: 1,
  b: {
    c: 2
  }
}

const obj2 = Object.assign({}, obj1)

obj2.b.c = 999

console.log(obj1.b.c)  // 999 ❌（obj1 也被修改了！）
console.log(obj2.b.c)  // 999

// 原因：obj1.b 和 obj2.b 指向同一个对象
console.log(obj1.b === obj2.b)  // true
```

### 2.2 会修改目标对象

```javascript
// ⚠️ 第一个参数会被修改
const target = { a: 1 }
Object.assign(target, { b: 2 })

console.log(target)  // { a: 1, b: 2 }（被修改了）

// ✅ 如果不想修改原对象，第一个参数传空对象
const original = { a: 1 }
const result = Object.assign({}, original, { b: 2 })

console.log(original)  // { a: 1 }（未修改）
console.log(result)    // { a: 1, b: 2 }
```

### 2.3 属性覆盖

```javascript
// 后面的源对象会覆盖前面的同名属性
const result = Object.assign(
  {},
  { a: 1, b: 2 },
  { b: 3, c: 4 },
  { c: 5 }
)

console.log(result)  // { a: 1, b: 3, c: 5 }
```

## 三、常见用法

### 3.1 克隆对象（浅拷贝）

```javascript
const obj = { a: 1, b: 2 }
const clone = Object.assign({}, obj)

// 等价于
const clone = { ...obj }  // ✅ 推荐：更简洁
```

### 3.2 合并对象

```javascript
const defaults = { timeout: 1000, retries: 3 }
const options = { timeout: 2000 }

const config = Object.assign({}, defaults, options)
// { timeout: 2000, retries: 3 }

// 等价于
const config = { ...defaults, ...options }  // ✅ 推荐
```

### 3.3 添加属性

```javascript
const user = { name: 'React' }
Object.assign(user, { age: 18, role: 'admin' })

console.log(user)  // { name: 'React', age: 18, role: 'admin' }

// 等价于
const user = { name: 'React' }
const newUser = { ...user, age: 18, role: 'admin' }  // ✅ 推荐：不修改原对象
```

## 四、易错点

### 4.1 深拷贝问题

```javascript
// ❌ 不能深拷贝
const obj = {
  a: 1,
  b: { c: 2 },
  d: [3, 4],
  e: new Date(),
  f: () => {}
}

const copy = Object.assign({}, obj)

// 嵌套对象和数组都是浅拷贝
console.log(obj.b === copy.b)  // true
console.log(obj.d === copy.d)  // true

// ✅ 深拷贝方案
// 方案1：JSON（简单但有限制）
const deepCopy1 = JSON.parse(JSON.stringify(obj))  // ⚠️ 会丢失函数、Date 等

// 方案2：递归
function deepClone(obj) {
  if (obj === null || typeof obj !== 'object') return obj
  if (obj instanceof Date) return new Date(obj)
  if (obj instanceof Array) return obj.map(item => deepClone(item))
  
  const cloned = {}
  for (const key in obj) {
    if (obj.hasOwnProperty(key)) {
      cloned[key] = deepClone(obj[key])
    }
  }
  return cloned
}

// 方案3：第三方库
import _ from 'lodash'
const deepCopy3 = _.cloneDeep(obj)
```

### 4.2 null 和 undefined

```javascript
// null 和 undefined 作为源对象会被忽略
const result = Object.assign({ a: 1 }, null, undefined, { b: 2 })
console.log(result)  // { a: 1, b: 2 }

// ❌ null 作为目标对象会报错
Object.assign(null, { a: 1 })  // TypeError!
```

### 4.3 数组处理

```javascript
// 数组会被当作对象处理（不推荐用 Object.assign 处理数组）
const arr1 = [1, 2, 3]
const arr2 = [4, 5]

const result = Object.assign([], arr1, arr2)
console.log(result)  // [4, 5, 3]（不是 [1, 2, 3, 4, 5]）

// 原因：
// arr1 → { 0: 1, 1: 2, 2: 3 }
// arr2 → { 0: 4, 1: 5 }
// 合并 → { 0: 4, 1: 5, 2: 3 }

// ✅ 数组合并用 concat 或展开运算符
const result = [...arr1, ...arr2]  // [1, 2, 3, 4, 5]
const result = arr1.concat(arr2)   // [1, 2, 3, 4, 5]
```

### 4.4 只复制可枚举属性

```javascript
const obj = Object.create(
  { inherited: 1 },  // 原型上的属性
  {
    own: {
      value: 2,
      enumerable: true
    },
    hidden: {
      value: 3,
      enumerable: false  // 不可枚举
    }
  }
)

const copy = Object.assign({}, obj)

console.log(copy)  // { own: 2 }（只复制了可枚举的自有属性）
console.log(copy.inherited)  // undefined（不复制继承的属性）
console.log(copy.hidden)     // undefined（不复制不可枚举属性）
```

## 五、React 中的应用

```javascript
// 1. setState 合并状态（类组件）
this.setState(prevState => 
  Object.assign({}, prevState, { count: prevState.count + 1 })
)

// 现代写法（推荐）
this.setState(prevState => ({ 
  ...prevState, 
  count: prevState.count + 1 
}))

// 2. Redux reducer 返回新状态
function reducer(state, action) {
  switch (action.type) {
    case 'UPDATE':
      return Object.assign({}, state, { value: action.payload })
    
    // 现代写法（推荐）
    case 'UPDATE':
      return { ...state, value: action.payload }
  }
}

// 3. 合并 props 和 defaultProps
const defaultProps = { size: 'medium', color: 'blue' }
const userProps = { color: 'red' }

const finalProps = Object.assign({}, defaultProps, userProps)
// { size: 'medium', color: 'red' }

// React 内部就是这样处理 defaultProps 的
```

## 六、手写实现

```javascript
// 简化版 Object.assign 实现
Object.myAssign = function(target, ...sources) {
  // 1. 目标对象不能是 null 或 undefined
  if (target == null) {
    throw new TypeError('Cannot convert undefined or null to object')
  }
  
  // 2. 转为对象
  const to = Object(target)
  
  // 3. 遍历所有源对象
  for (let i = 0; i < sources.length; i++) {
    const source = sources[i]
    
    // 忽略 null 和 undefined
    if (source == null) continue
    
    // 4. 复制可枚举属性
    for (const key in source) {
      if (source.hasOwnProperty(key)) {
        to[key] = source[key]
      }
    }
  }
  
  return to
}

// 测试
const result = Object.myAssign({ a: 1 }, { b: 2 }, { c: 3 })
console.log(result)  // { a: 1, b: 2, c: 3 }
```

## 七、面试要点

### Q1: Object.assign 是深拷贝还是浅拷贝？

**浅拷贝**。嵌套对象只复制引用，不复制值。

```javascript
const obj = { a: { b: 1 } }
const copy = Object.assign({}, obj)

console.log(obj.a === copy.a)  // true（指向同一个对象）
```

### Q2: Object.assign 和展开运算符的区别？

```javascript
// 功能基本相同，都是浅拷贝
const result1 = Object.assign({}, obj1, obj2)
const result2 = { ...obj1, ...obj2 }

// 区别：
// 1. 语法：展开运算符更简洁
// 2. 性能：现代浏览器中展开运算符略快
// 3. 可读性：展开运算符更直观
// 4. Object.assign 会修改第一个参数，展开运算符不会

// ✅ 推荐使用展开运算符
```

### Q3: 如何深拷贝一个对象？

```javascript
// 方案1：JSON（简单但有限制）
const deepCopy = JSON.parse(JSON.stringify(obj))
// 限制：无法处理函数、Date、RegExp、undefined、循环引用

// 方案2：递归实现
function deepClone(obj, map = new WeakMap()) {
  if (obj === null || typeof obj !== 'object') return obj
  if (map.has(obj)) return map.get(obj)  // 处理循环引用
  
  const cloned = Array.isArray(obj) ? [] : {}
  map.set(obj, cloned)
  
  for (const key in obj) {
    if (obj.hasOwnProperty(key)) {
      cloned[key] = deepClone(obj[key], map)
    }
  }
  
  return cloned
}

// 方案3：第三方库
import { cloneDeep } from 'lodash'
const deepCopy = cloneDeep(obj)

// 方案4：structuredClone（现代浏览器）
const deepCopy = structuredClone(obj)
```

### Q4: Object.assign 在 React 中的应用？

1. **合并状态**：`Object.assign({}, prevState, newState)`
2. **合并配置**：`Object.assign({}, defaults, options)`
3. **Redux reducer**：返回新的 state 对象
4. **现在推荐用展开运算符代替**

## 八、总结

### 核心记忆点

1. **浅拷贝**：嵌套对象只复制引用
2. **修改 target**：第一个参数会被修改
3. **属性覆盖**：后面覆盖前面
4. **推荐替代**：现在更推荐用展开运算符 `{ ...obj }`

### 使用建议

```javascript
// ❌ 不推荐（老写法）
const newObj = Object.assign({}, obj1, obj2)

// ✅ 推荐（现代写法）
const newObj = { ...obj1, ...obj2 }
```

### 适用场景

- ✅ 浅拷贝对象
- ✅ 合并多个对象
- ❌ 深拷贝（用其他方案）
- ❌ 数组合并（用 concat 或展开运算符）
