## 1. 什么是柯里化

### 1.1 定义

**柯里化（Currying）**：将一个接受多个参数的函数，转换为一系列只接受单个参数的函数。

```javascript
// 普通函数
function add(a, b, c) {
  return a + b + c;
}
add(1, 2, 3); // 6

// 柯里化后
function curriedAdd(a) {
  return function(b) {
    return function(c) {
      return a + b + c;
    };
  };
}
curriedAdd(1)(2)(3); // 6
```

---

### 1.2 核心概念

**转换过程**：
```
f(a, b, c) → f(a)(b)(c)

或

f(a, b, c) → f(a)(b, c) → f(a)(b)(c)
```

**特点**：
- ✅ 参数复用
- ✅ 延迟执行
- ✅ 函数组合
- ✅ 代码可读性

---

### 1.3 简单示例

```javascript
// 示例 1：简单加法
const add = a => b => c => a + b + c;
add(1)(2)(3); // 6

// 示例 2：字符串拼接
const greet = greeting => name => `${greeting}, ${name}!`;
const sayHello = greet('Hello');
sayHello('Alice'); // "Hello, Alice!"
sayHello('Bob');   // "Hello, Bob!"

// 示例 3：日志函数
const log = level => message => {
  console.log(`[${level}] ${message}`);
};
const error = log('ERROR');
error('Something went wrong'); // [ERROR] Something went wrong
error('Network failure');      // [ERROR] Network failure
```

## 2. 柯里化原理

### 2.1 闭包与参数累积

```javascript
function curry(fn) {
  // 1. 保存原始函数的参数个数
  const arity = fn.length;
  
  // 2. 返回柯里化函数
  return function curried(...args) {
    // 3. 如果参数足够，执行原函数
    if (args.length >= arity) {
      return fn.apply(this, args);
    }
    
    // 4. 参数不足，返回新函数继续收集参数
    return function(...nextArgs) {
      return curried.apply(this, args.concat(nextArgs));
    };
  };
}

// 使用
function add(a, b, c) {
  return a + b + c;
}

const curriedAdd = curry(add);
console.log(curriedAdd(1)(2)(3));     // 6
console.log(curriedAdd(1, 2)(3));     // 6
console.log(curriedAdd(1)(2, 3));     // 6
console.log(curriedAdd(1, 2, 3));     // 6
```

---

### 2.2 执行流程分析

```javascript
const curriedAdd = curry(add);

// 调用 curriedAdd(1)
curriedAdd(1)
  ↓
args = [1], arity = 3
args.length (1) < arity (3)
  ↓
返回新函数：function(...nextArgs) {
  return curried(1, ...nextArgs);
}

// 调用 curriedAdd(1)(2)
curriedAdd(1)(2)
  ↓
args = [1, 2], arity = 3
args.length (2) < arity (3)
  ↓
返回新函数：function(...nextArgs) {
  return curried(1, 2, ...nextArgs);
}

// 调用 curriedAdd(1)(2)(3)
curriedAdd(1)(2)(3)
  ↓
args = [1, 2, 3], arity = 3
args.length (3) >= arity (3)
  ↓
执行：add(1, 2, 3) → 6
```

### 2.3 参数占位符

```javascript
// 支持占位符的柯里化
const _ = Symbol('placeholder');

function curryWithPlaceholder(fn) {
  const arity = fn.length;
  
  return function curried(...args) {
    // 计算实际参数个数（排除占位符）
    const realArgs = args.filter(arg => arg !== _);
    
    if (realArgs.length >= arity && !args.includes(_)) {
      return fn.apply(this, args);
    }
    
    return function(...nextArgs) {
      // 替换占位符
      const newArgs = args.map(arg => 
        arg === _ ? nextArgs.shift() : arg
      );
      // 追加剩余参数
      return curried(...newArgs, ...nextArgs);
    };
  };
}

// 使用
const curriedAdd = curryWithPlaceholder(add);

// 正常使用
curriedAdd(1)(2)(3); // 6

// 使用占位符
curriedAdd(_, 2)(1, 3); // 6
curriedAdd(_, _, 3)(1)(2); // 6
curriedAdd(1, _, 3)(2); // 6
```

## 3. 手写柯里化函数

### 3.1 基础版本

```javascript
/**
 * 柯里化函数 - 基础版
 * @param {Function} fn - 原始函数
 * @returns {Function} - 柯里化后的函数
 */
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    }
    return function(...nextArgs) {
      return curried.apply(this, args.concat(nextArgs));
    };
  };
}

// 测试
function sum(a, b, c, d) {
  return a + b + c + d;
}

const curriedSum = curry(sum);
console.log(curriedSum(1)(2)(3)(4));       // 10
console.log(curriedSum(1, 2)(3)(4));       // 10
console.log(curriedSum(1)(2, 3, 4));       // 10
```

---

### 3.2 增强版（支持多种调用方式）

```javascript
/**
 * 柯里化函数 - 增强版
 * @param {Function} fn - 原始函数
 * @param {number} arity - 参数个数（可选，默认为 fn.length）
 * @returns {Function} - 柯里化后的函数
 */
function curry(fn, arity = fn.length) {
  return function curried(...args) {
    // 参数足够，执行函数
    if (args.length >= arity) {
      return fn.apply(this, args);
    }
    
    // 参数不足，返回新函数
    return function(...nextArgs) {
      return curried.apply(this, [...args, ...nextArgs]);
    };
  };
}

// 测试
function multiply(a, b, c) {
  return a * b * c;
}

const curriedMultiply = curry(multiply);

// 支持多种调用方式
console.log(curriedMultiply(2)(3)(4));     // 24
console.log(curriedMultiply(2, 3)(4));     // 24
console.log(curriedMultiply(2)(3, 4));     // 24
console.log(curriedMultiply(2, 3, 4));     // 24
```

### 3.3 完整版（支持占位符）

```javascript
/**
 * 柯里化函数 - 完整版
 * @param {Function} fn - 原始函数
 * @returns {Function} - 柯里化后的函数
 */
function curry(fn) {
  const arity = fn.length;
  const placeholder = curry.placeholder;
  
  return function curried(...args) {
    // 过滤占位符，计算实际参数个数
    const realArgs = args.filter(arg => arg !== placeholder);
    
    // 参数足够且无占位符，执行函数
    if (realArgs.length >= arity && !args.includes(placeholder)) {
      return fn.apply(this, args.slice(0, arity));
    }
    
    // 参数不足，返回新函数
    return function(...nextArgs) {
      // 替换占位符
      const newArgs = [];
      let nextIndex = 0;
      
      for (let i = 0; i < args.length; i++) {
        if (args[i] === placeholder) {
          newArgs.push(nextArgs[nextIndex++] || placeholder);
        } else {
          newArgs.push(args[i]);
        }
      }
      
      // 追加剩余参数
      while (nextIndex < nextArgs.length) {
        newArgs.push(nextArgs[nextIndex++]);
      }
      
      return curried(...newArgs);
    };
  };
}

// 定义占位符
curry.placeholder = Symbol('placeholder');
const _ = curry.placeholder;

// 测试
function test(a, b, c, d) {
  return `a:${a}, b:${b}, c:${c}, d:${d}`;
}

const curriedTest = curry(test);

console.log(curriedTest(1, 2, 3, 4));        // a:1, b:2, c:3, d:4
console.log(curriedTest(_, 2)(1, 3, 4));     // a:1, b:2, c:3, d:4
console.log(curriedTest(_, _, 3)(1)(2, 4));  // a:1, b:2, c:3, d:4
console.log(curriedTest(1, _, _, 4)(2, 3));  // a:1, b:2, c:3, d:4
```

### 3.4 无限柯里化

```javascript
/**
 * 无限柯里化 - 没有固定参数个数
 * 调用 valueOf/toString 时触发计算
 */
function infiniteCurry(fn) {
  const args = [];
  
  function curried(...newArgs) {
    if (newArgs.length === 0) {
      // 无参数调用，触发计算
      return fn(...args);
    }
    
    // 累积参数
    args.push(...newArgs);
    return curried;
  }
  
  // 支持隐式转换
  curried.valueOf = curried.toString = function() {
    return fn(...args);
  };
  
  return curried;
}

// 示例 1：求和
function sum(...nums) {
  return nums.reduce((acc, num) => acc + num, 0);
}

const add = infiniteCurry(sum);
console.log(add(1)(2)(3).valueOf());  // 6
console.log(add(1, 2)(3, 4).valueOf()); // 10
console.log(+add(1)(2)(3));           // 6（隐式转换）

// 示例 2：链式调用
function multiply(...nums) {
  return nums.reduce((acc, num) => acc * num, 1);
}

const mul = infiniteCurry(multiply);
console.log(mul(2)(3)(4).valueOf());  // 24
```

## 5. 柯里化 vs 偏函数

### 5.1 概念对比

| 特性 | 柯里化（Currying）| 偏函数（Partial Application）|
|------|------------------|------------------------------|
| **定义** | 将多参数函数转为单参数函数序列 | 固定部分参数，返回新函数 |
| **调用方式** | `f(a)(b)(c)` | `f(a, b)(c)` |
| **参数数量** | 每次只接受一个参数 | 每次可接受多个参数 |
| **灵活性** | 低（严格单参数）| 高（可固定任意数量参数）|
| **用途** | 函数组合、参数复用 | 简化函数调用 |

---

### 5.2 柯里化示例

```javascript
// 柯里化：每次只接受一个参数
const curriedAdd = a => b => c => a + b + c;

curriedAdd(1)(2)(3);  // ✓ 6
curriedAdd(1, 2)(3);  // ✗ NaN（不支持多参数）
```

### 5.4 实现偏函数

```javascript
console.log(sayHelloToAlice('!'));        // Hello, Alice!
```

### 5.5 偏函数支持占位符

```javascript
/**
 * 偏函数 - 支持占位符
 */
const _ = Symbol('placeholder');

function partial(fn, ...args) {
  return function partialApplied(...restArgs) {
    const newArgs = [];
    let restIndex = 0;
    
    // 替换占位符
    for (let i = 0; i < args.length; i++) {
      if (args[i] === _) {
        newArgs.push(restArgs[restIndex++]);
      } else {
        newArgs.push(args[i]);
      }
    }
    
    // 追加剩余参数
    while (restIndex < restArgs.length) {
      newArgs.push(restArgs[restIndex++]);
    }
    
    return fn.apply(this, newArgs);
  };
}

// 示例
function subtract(a, b, c) {
  return a - b - c;
}

const sub = partial(subtract, _, 2, _);
console.log(sub(10, 3));  // 10 - 2 - 3 = 5
```

## 6. 实战案例

### 6.1 React Hooks

```javascript
// 场景：useCallback 与柯里化
import { useState, useCallback } from 'react';

function TodoList() {
  const [todos, setTodos] = useState([]);
  
  // 柯里化的事件处理器
  const handleToggle = useCallback((id) => () => {
    setTodos(todos => 
      todos.map(todo => 
        todo.id === id ? { ...todo, done: !todo.done } : todo
      )
    );
  }, []);
  
  const handleDelete = useCallback((id) => () => {
    setTodos(todos => todos.filter(todo => todo.id !== id));
  }, []);
  
  return (
    <ul>
      {todos.map(todo => (
        <li key={todo.id}>
          <input 
            type="checkbox" 
            checked={todo.done}
            onChange={handleToggle(todo.id)}  {/* 柯里化 */}
          />
          <span>{todo.text}</span>
          <button onClick={handleDelete(todo.id)}>Delete</button>
        </li>
      ))}
    </ul>
  );
}
```

### 6.3 表单验证

```javascript
// 场景：复杂表单验证
const curry = fn => {
  const arity = fn.length;
  return function curried(...args) {
    if (args.length >= arity) {
      return fn(...args);
    }
    return (...nextArgs) => curried(...args, ...nextArgs);
  };
};

// 基础验证函数
const createValidator = curry((rule, message, value) => {
  if (!rule(value)) {
    return { valid: false, message };
  }
  return { valid: true };
});

// 验证规则
const required = createValidator(
  v => v !== null && v !== undefined && v !== '',
  'This field is required'
);

const minLength = min => createValidator(
  v => v.length >= min,
  `Minimum length is ${min}`
);

const maxLength = max => createValidator(
  v => v.length <= max,
  `Maximum length is ${max}`
);

const pattern = (regex, message) => createValidator(
  v => regex.test(v),
  message
);

const email = pattern(
  /^[^\s@]+@[^\s@]+\.[^\s@]+$/,
  'Invalid email format'
);

// 组合多个验证器
function composeValidators(...validators) {
  return value => {
    for (const validator of validators) {
      const result = validator(value);
      if (!result.valid) {
        return result;
      }
    }
    return { valid: true };
  };
}

// 使用
const validateUsername = composeValidators(
  required,
  minLength(3),
  maxLength(20),
  pattern(/^[a-zA-Z0-9_]+$/, 'Only alphanumeric and underscore allowed')
);

const validateEmail = composeValidators(
  required,
  email
);

const validatePassword = composeValidators(
  required,
  minLength(8),
  pattern(/[A-Z]/, 'Must contain at least one uppercase letter'),
  pattern(/[a-z]/, 'Must contain at least one lowercase letter'),
  pattern(/[0-9]/, 'Must contain at least one number')
);

// 测试
console.log(validateUsername('ab'));        // { valid: false, message: 'Minimum length is 3' }
console.log(validateUsername('alice123'));  // { valid: true }
console.log(validateEmail('alice'));        // { valid: false, message: 'Invalid email format' }
console.log(validateEmail('alice@example.com')); // { valid: true }
```

### 6.5 管道与组合

```javascript
// 管道函数（从左到右）
const pipe = (...fns) => x => fns.reduce((v, f) => f(v), x);

// 组合函数（从右到左）
const compose = (...fns) => x => fns.reduceRight((v, f) => f(v), x);

// 基础函数（柯里化）
const add = a => b => a + b;
const multiply = a => b => a * b;
const divide = a => b => b / a;
const subtract = a => b => b - a;

// 使用 pipe
const calculate1 = pipe(
  add(10),        // x + 10
  multiply(2),    // (x + 10) * 2
  subtract(5)     // ((x + 10) * 2) - 5
);

console.log(calculate1(5));  // (5 + 10) * 2 - 5 = 25

// 使用 compose
const calculate2 = compose(
  subtract(5),    // 最后执行
  multiply(2),    // 第二执行
  add(10)         // 最先执行
);

console.log(calculate2(5));  // (5 + 10) * 2 - 5 = 25

// 实际应用：数据处理
const users = [
  { id: 1, name: 'Alice', age: 25, active: true },
  { id: 2, name: 'Bob', age: 30, active: false },
  { id: 3, name: 'Charlie', age: 35, active: true }
];

const filter = predicate => arr => arr.filter(predicate);
const map = fn => arr => arr.map(fn);
const take = n => arr => arr.slice(0, n);

const getActiveUserNames = pipe(
  filter(user => user.active),
  map(user => user.name),
  take(2)
);

console.log(getActiveUserNames(users));  // ['Alice', 'Charlie']
```

## 7. 性能优化

### 7.1 缓存优化

```javascript
/**
 * 带缓存的柯里化
 */
function memoizedCurry(fn) {
  const cache = new Map();
  
  return function curried(...args) {
    const key = JSON.stringify(args);
    
    if (cache.has(key)) {
      console.log('Cache hit');
      return cache.get(key);
    }
    
    if (args.length >= fn.length) {
      const result = fn.apply(this, args);
      cache.set(key, result);
      return result;
    }
    
    return function(...nextArgs) {
      return curried.apply(this, args.concat(nextArgs));
    };
  };
}

// 测试
function expensiveCalculation(a, b, c) {
  console.log('Calculating...');
  return a * b * c;
}

const cached = memoizedCurry(expensiveCalculation);

console.log(cached(2)(3)(4));  // Calculating... 24
console.log(cached(2)(3)(4));  // Cache hit, 24
```

## 8. 常见问题

### 8.1 为什么需要柯里化？

**优点**：
1. **参数复用**：固定部分参数，创建新函数
2. **延迟执行**：分步传参，最后统一执行
3. **函数组合**：更容易组合函数
4. **代码可读性**：语义更清晰

**缺点**：
1. **性能开销**：闭包 + 多次函数调用
2. **调试困难**：堆栈信息不直观
3. **理解成本**：新手不易理解

---

### 8.2 柯里化的局限性

```javascript
// 问题 1：无法处理可变参数
function sum(...args) {
  return args.reduce((a, b) => a + b, 0);
}

const curriedSum = curry(sum);
// sum.length === 0，柯里化失效

// 解决：手动指定参数个数
const curriedSum = curry(sum, 3);
curriedSum(1)(2)(3);  // 6

// 问题 2：this 绑定
const obj = {
  name: 'Alice',
  greet: function(greeting) {
    return `${greeting}, ${this.name}`;
  }
};

const curriedGreet = curry(obj.greet);
curriedGreet('Hello');  // this 丢失

// 解决：使用 bind 或箭头函数
const curriedGreet = curry(obj.greet.bind(obj));
curriedGreet('Hello');  // Hello, Alice
```

---

### 8.3 如何选择使用？

**适合柯里化的场景**：
- 函数式编程
- 参数复用
- 函数组合
- 配置函数

**不适合柯里化的场景**：
- 性能敏感的代码
- 可变参数函数
- 需要保留 this 的方法

## 9. 面试高频问答

### Q1: 什么是函数柯里化？

**答案**：

函数柯里化是将一个接受多个参数的函数，转换为一系列只接受单个参数的函数的技术。

```javascript
// 普通函数
function add(a, b, c) {
  return a + b + c;
}
add(1, 2, 3);  // 6

// 柯里化后
const curriedAdd = a => b => c => a + b + c;
curriedAdd(1)(2)(3);  // 6
```

**核心特点**：
- 参数逐个传入
- 每次返回新函数
- 最后返回结果

### Q2: 手写一个柯里化函数

**答案**：

```javascript
function curry(fn) {
  return function curried(...args) {
    // 参数足够，执行原函数
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    }
    
    // 参数不足，返回新函数
    return function(...nextArgs) {
      return curried.apply(this, args.concat(nextArgs));
    };
  };
}

// 测试
function sum(a, b, c) {
  return a + b + c;
}

const curriedSum = curry(sum);
console.log(curriedSum(1)(2)(3));     // 6
console.log(curriedSum(1, 2)(3));     // 6
console.log(curriedSum(1)(2, 3));     // 6
```

### Q3: 柯里化和偏函数有什么区别？

**答案**：

| 特性 | 柯里化 | 偏函数 |
|------|--------|--------|
| **定义** | 多参数 → 单参数序列 | 固定部分参数 |
| **调用** | `f(a)(b)(c)` | `f(a, b)(c)` |
| **灵活性** | 低（严格单参数）| 高（可多参数）|

```javascript
// 柯里化
const curriedAdd = a => b => c => a + b + c;
curriedAdd(1)(2)(3);  // 6

// 偏函数
function partial(fn, ...args) {
  return (...restArgs) => fn(...args, ...restArgs);
}

const add1 = partial(add, 1);
add1(2, 3);  // 6
```

### Q6: 如何实现支持占位符的柯里化？

**答案**：

```javascript
const _ = Symbol('placeholder');

function curry(fn) {
  const arity = fn.length;
  
  return function curried(...args) {
    const realArgs = args.filter(arg => arg !== _);
    
    if (realArgs.length >= arity && !args.includes(_)) {
      return fn.apply(this, args);
    }
    
    return function(...nextArgs) {
      const newArgs = [];
      let nextIndex = 0;
      
      for (let i = 0; i < args.length; i++) {
        if (args[i] === _) {
          newArgs.push(nextArgs[nextIndex++] || _);
        } else {
          newArgs.push(args[i]);
        }
      }
      
      while (nextIndex < nextArgs.length) {
        newArgs.push(nextArgs[nextIndex++]);
      }
      
      return curried(...newArgs);
    };
  };
}

// 使用
const add = (a, b, c) => a + b + c;
const curriedAdd = curry(add);

curriedAdd(_, 2)(1, 3);      // 6
curriedAdd(_, _, 3)(1)(2);   // 6
```

### Q8: 如何实现无限柯里化？

**答案**：

```javascript
function infiniteCurry(fn) {
  const args = [];
  
  function curried(...newArgs) {
    if (newArgs.length === 0) {
      return fn(...args);
    }
    args.push(...newArgs);
    return curried;
  }
  
  // 支持隐式转换
  curried.valueOf = curried.toString = function() {
    return fn(...args);
  };
  
  return curried;
}

// 使用
function sum(...nums) {
  return nums.reduce((a, b) => a + b, 0);
}

const add = infiniteCurry(sum);
console.log(add(1)(2)(3).valueOf());  // 6
console.log(+add(1, 2)(3, 4));        // 10
```

### Q9: 柯里化和闭包的关系？

**答案**：

柯里化**依赖闭包**实现参数的保存和累积。

```javascript
function curry(fn) {
  return function curried(...args) {  // 外层函数
    if (args.length >= fn.length) {
      return fn(...args);
    }
    
    // 内层函数形成闭包，访问外层的 args
    return function(...nextArgs) {
      return curried(...args, ...nextArgs);  // 访问外层 args
    };
  };
}

// 原理
const curriedAdd = curry((a, b, c) => a + b + c);
const step1 = curriedAdd(1);  // step1 闭包保存 args = [1]
const step2 = step1(2);       // step2 闭包保存 args = [1, 2]
const result = step2(3);      // 执行：1 + 2 + 3 = 6
```

**关键点**：
- 每次调用返回新函数
- 新函数通过闭包访问之前的参数
- 参数累积直到满足条件

### Q10: 如何优化柯里化的性能？

**答案**：

**方式 1：使用记忆化**
```javascript
function memoizedCurry(fn) {
  const cache = new Map();
  
  return function curried(...args) {
    const key = JSON.stringify(args);
    
    if (cache.has(key)) {
      return cache.get(key);
    }
    
    let result;
    if (args.length >= fn.length) {
      result = fn(...args);
    } else {
      result = (...nextArgs) => curried(...args, ...nextArgs);
    }
    
    cache.set(key, result);
    return result;
  };
}
```

**方式 2：减少不必要的柯里化**
```javascript
// ❌ 过度柯里化
const add = a => b => c => d => e => a + b + c + d + e;

// ✅ 合理分组
const add = (a, b) => (c, d) => e => a + b + c + d + e;
```

**方式 3：在非关键路径使用**
```javascript
// 关键路径：直接调用
for (let i = 0; i < 1000000; i++) {
  add(1, 2, 3);
}

// 非关键路径：柯里化
const createLogger = level => module => message => {
  console.log(`[${level}] [${module}] ${message}`);
};
```
