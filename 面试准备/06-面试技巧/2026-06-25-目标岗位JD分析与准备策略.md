#### Q2: 如何设计一个高性能的虚拟列表？
**参考答案：**
```javascript
// 核心思路
1. 只渲染可见区域的元素（visibleStart → visibleEnd）
2. 使用padding撑起总高度，保持滚动条正确
3. 监听scroll事件，动态更新可见区域
4. 使用IntersectionObserver优化滚动监听

// 实现要点
function VirtualList({ items, itemHeight, containerHeight }) {
  const [scrollTop, setScrollTop] = useState(0);
  
  // 计算可见区域
  const visibleStart = Math.floor(scrollTop / itemHeight);
  const visibleEnd = Math.ceil((scrollTop + containerHeight) / itemHeight);
  const visibleItems = items.slice(visibleStart, visibleEnd);
  
  // 计算偏移
  const offsetY = visibleStart * itemHeight;
  const totalHeight = items.length * itemHeight;
  
  return (
    <div
      style={{ height: containerHeight, overflow: 'auto' }}
      onScroll={(e) => setScrollTop(e.target.scrollTop)}
    >
      <div style={{ height: totalHeight, position: 'relative' }}>
        <div style={{ transform: `translateY(${offsetY}px)` }}>
          {visibleItems.map((item, index) => (
            <div key={visibleStart + index} style={{ height: itemHeight }}>
              {item}
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}

// 优化点
- 使用transform代替top（避免重排）
- 使用requestAnimationFrame节流滚动事件
- 使用React.memo缓存列表项
- 动态高度列表：维护高度缓存Map
```

**实战数据：**
> "在YChat客服系统中，我实现的虚拟列表支持10万+消息，滚动帧率稳定60fps，内存占用从1.2GB降到600MB，用户体验显著提升。"

#### Q3: 如何实现一个高性能的Canvas图表？
**参考答案：**
```javascript
// 优化策略
1. 离屏Canvas（OffscreenCanvas）
2. 分层渲染（静态层 + 动态层）
3. 脏矩形（Dirty Rectangle）优化
4. requestAnimationFrame动画循环
5. WebWorker处理数据计算

// 实现示例
class Chart {
  constructor(canvas) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d');
    this.offscreenCanvas = document.createElement('canvas');
    this.offscreenCtx = this.offscreenCanvas.getContext('2d');
  }
  
  // 分层渲染
  renderStatic() {
    // 离屏渲染静态内容（坐标轴、网格）
    this.offscreenCtx.clearRect(0, 0, width, height);
    this.drawAxis(this.offscreenCtx);
    this.drawGrid(this.offscreenCtx);
  }
  
  renderDynamic() {
    // 主Canvas渲染动态内容
    this.ctx.clearRect(0, 0, width, height);
    this.ctx.drawImage(this.offscreenCanvas, 0, 0); // 复制静态层
    this.drawData(this.ctx); // 绘制数据
  }
  
  // 脏矩形优化
  update(dirtyRect) {
    const { x, y, width, height } = dirtyRect;
    this.ctx.clearRect(x, y, width, height);
    // 只重绘局部区域
  }
}
```

   - Result：最终成果

   - 最佳实践：总结方法论和通用模式
