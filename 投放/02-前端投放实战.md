```tsx
  const [previewImage, setPreviewImage] = useState('')
  
  // 上传前校验
  const beforeUpload = (file: File) => {
    // 1. 尺寸校验
    return new Promise<void>((resolve, reject) => {
      const img = new Image()
      img.onload = () => {
        const isValidSize = 
          img.width >= 600 && img.width <= 1200 &&
          img.height >= 400 && img.height <= 800
        
        if (!isValidSize) {
          message.error('图片尺寸不符合要求（600x400 - 1200x800）')
          reject()
        } else {
          resolve()
        }
      }
      img.src = URL.createObjectURL(file)
    })
  }
  
  // 自定义上传
  const customUpload = async (options: any) => {
    const { file, onProgress, onSuccess, onError } = options
    
    try {
      // 1. 获取上传凭证
      const { uploadUrl, key } = await getUploadToken()
      
      // 2. 上传到 OSS
      await axios.put(uploadUrl, file, {
        headers: { 'Content-Type': file.type },
        onUploadProgress: (e) => {
          onProgress({ percent: (e.loaded / e.total) * 100 })
        }
      })
      
      // 3. 回调成功
      onSuccess({ url: `https://cdn.example.com/${key}` })
    } catch (error) {
      onError(error)
    }
  }
  
  return (
    <div className="creative-upload">
      <Upload
        listType="picture-card"
        fileList={fileList}
        beforeUpload={beforeUpload}
        customRequest={customUpload}
        onChange={({ fileList }) => setFileList(fileList)}
        onPreview={(file) => setPreviewImage(file.url!)}
      >
        {fileList.length < 5 && (
          <div>
            <PlusOutlined />
            <div>上传图片</div>
          </div>

        type="info"
```

```typescript
// WebSocket 连接管理
class WebSocketClient {
  private ws: WebSocket | null = null
  private reconnectTimer: NodeJS.Timeout | null = null
  private heartbeatTimer: NodeJS.Timeout | null = null
  
  connect(url: string, onMessage: (data: any) => void) {
    this.ws = new WebSocket(url)
    
    this.ws.onopen = () => {
      console.log('WebSocket connected')
      this.startHeartbeat()
    }
    
    this.ws.onmessage = (event) => {
      const data = JSON.parse(event.data)
      onMessage(data)
    }
    
    this.ws.onclose = () => {
      console.log('WebSocket disconnected')
      this.reconnect(url, onMessage)
    }
    
    this.ws.onerror = (error) => {
      console.error('WebSocket error:', error)
    }
  }
  
  private startHeartbeat() {
    this.heartbeatTimer = setInterval(() => {
      if (this.ws?.readyState === WebSocket.OPEN) {
        this.ws.send(JSON.stringify({ type: 'ping' }))
      }
    }, 30000) // 每30秒发送一次心跳
  }
  
  private reconnect(url: string, onMessage: (data: any) => void) {
    this.reconnectTimer = setTimeout(() => {
      console.log('Reconnecting...')
      this.connect(url, onMessage)
    }, 5000) // 5秒后重连
  }
  
  disconnect() {
    if (this.reconnectTimer) clearTimeout(this.reconnectTimer)
    if (this.heartbeatTimer) clearInterval(this.heartbeatTimer)
    this.ws?.close()
  }
}

// 数字动画组件
const AnimatedNumber: React.FC<{ value: number; decimals?: number }> = ({ 
  value, 
  decimals = 0 
}) => {
  const [displayValue, setDisplayValue] = useState(0)
  
  useEffect(() => {
    const duration = 500 // 动画时长
    const steps = 30
    const increment = (value - displayValue) / steps
    let current = displayValue
    
    const timer = setInterval(() => {
      current += increment
      if (Math.abs(current - value) < Math.abs(increment)) {
        setDisplayValue(value)
        clearInterval(timer)
      } else {
        setDisplayValue(current)
      }
    }, duration / steps)
    
    return () => clearInterval(timer)
  }, [value])
  
  return <>{displayValue.toFixed(decimals)}</>
}
```

```typescript
export const useSSE = (url: string) => {

      eventSource.close()
    }
    
    return () => eventSource.close()
  }, [url])
  
  return { data, error }
}
```

### 6.1 Zustand 状态管理

```typescript
import { create } from 'zustand'
import { devtools, persist } from 'zustand/middleware'

// 广告计划状态
interface CampaignStore {
  campaigns: Campaign[]
  selectedCampaign: Campaign | null
  filters: CampaignFilters
  
  fetchCampaigns: () => Promise<void>
  selectCampaign: (id: string) => void
  updateCampaign: (id: string, data: Partial<Campaign>) => Promise<void>
  deleteCampaign: (id: string) => Promise<void>
  setFilters: (filters: Partial<CampaignFilters>) => void
}

export const useCampaignStore = create<CampaignStore>()(
  devtools(
    persist(
      (set, get) => ({
        campaigns: [],
        selectedCampaign: null,
        filters: {},
        
        fetchCampaigns: async () => {
          const data = await api.getCampaigns(get().filters)
          set({ campaigns: data })
        },
        
        selectCampaign: (id) => {
          const campaign = get().campaigns.find(c => c.id === id)
          set({ selectedCampaign: campaign || null })
        },
        
        updateCampaign: async (id, data) => {
          await api.updateCampaign(id, data)
          set((state) => ({
            campaigns: state.campaigns.map(c =>
              c.id === id ? { ...c, ...data } : c
            )
          }))
        },
        
        deleteCampaign: async (id) => {
          await api.deleteCampaign(id)
          set((state) => ({
            campaigns: state.campaigns.filter(c => c.id !== id)
          }))
        },
        
        setFilters: (filters) => {
          set((state) => ({
            filters: { ...state.filters, ...filters }
          }))
          get().fetchCampaigns()
        }
      }),
      {
        name: 'campaign-store',
        partialize: (state) => ({ filters: state.filters })
      }
    )
  )
)
```

### 8.1 错误边界

```tsx
import { Component, ErrorInfo, ReactNode } from 'react'

interface Props {
  children: ReactNode
  fallback?: ReactNode
}

interface State {
  hasError: boolean
  error?: Error
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props)
    this.state = { hasError: false }
  }
  
  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error }
  }
  
  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('ErrorBoundary caught:', error, errorInfo)
    // 上报错误到监控平台
    reportError(error, errorInfo)
  }
  
  render() {
    if (this.state.hasError) {
      return this.props.fallback || (
        <Result
          status="error"
          title="页面加载失败"
          subTitle={this.state.error?.message}
          extra={
            <Button onClick={() => window.location.reload()}>
              刷新页面
            </Button>
          }
        />
      )
    }
    
    return this.props.children
  }
}
```

```typescript
import axios, { AxiosRequestConfig } from 'axios'

const request = axios.create({

// 请求拦截器

  }
```
